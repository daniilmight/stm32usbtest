
Option Explicit

' =====================================
' WORD PE v2.4
' =====================================

Private Type PEItem
    Code As String
    Num As Long
    Model As String
    Maker As String
    TokenId As Long
    NoteText As String
End Type

Private Type PERecord
    Ref As String
    Code As String
    TypeKey As String
    Group As String
    Model As String
    Maker As String
    Qty As Long

    TUBlockId As Long
    TUHeaderSeries As String

    NoteText As String
    NoteNo As Long
End Type

Private Const FIRST_ROWS As Long = 23
Private Const NEXT_ROWS As Long = 30
Private Const REF_PADDING As Double = 0.08
Private Const MIN_TU_ITEMS As Long = 4


' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Public Sub MakeWordPE()

    Dim xl As Object
    Dim book As Object
    Dim sh As Object

    Dim doc As Document
    Dim testDoc1 As Document
    Dim testDoc2 As Document

    Dim test1 As Table
    Dim test2 As Table
    Dim probe As Table

    Dim a As Variant
    Dim el() As PEItem
    Dim tmp As PEItem
    Dim rec() As PERecord

    Dim parts As Variant
    Dim part As Variant
    Dim piece As Variant

    Dim pages As Collection
    Dim pageRows As Collection
    Dim chunk As Collection
    Dim notes As Collection

    Dim templateFile As String
    Dim excelFile As String
    Dim outputFile As String
    Dim baseName As String

    Dim ref As String
    Dim code As String
    Dim endCode As String
    Dim firstRef As String
    Dim lastRef As String

    Dim model As String
    Dim maker As String
    Dim replacement As String
    Dim noteText As String

    Dim total As Long
    Dim recCount As Long
    Dim last As Long
    Dim token As Long

    Dim i As Long
    Dim j As Long
    Dim k As Long
    Dim num As Long
    Dim n1 As Long
    Dim n2 As Long
    Dim pos As Long

    Dim runEnd As Long
    Dim endIndex As Long
    Dim maxFinish As Long
    Dim limit As Long
    Dim freeRows As Long

    Dim found As Boolean

    On Error GoTo ErrorHandler


    ' =====================================
    ' ВЫБОР WORD-ШАБЛОНА
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите шаблон Word"
        .Filters.Clear
        .Filters.Add "Word", "*.doc;*.docx;*.docm"

        If .Show <> -1 Then Exit Sub

        templateFile = .SelectedItems(1)

    End With


    ' =====================================
    ' ВЫБОР EXCEL
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите перечень Excel"
        .Filters.Clear
        .Filters.Add "Excel", "*.xlsx;*.xlsm;*.xls"

        If .Show <> -1 Then Exit Sub

        excelFile = .SelectedItems(1)

    End With


    ' =====================================
    ' ЧТЕНИЕ EXCEL
    '
    ' A - обозначение
    ' B - наименование
    ' C - количество
    ' D - допускаемая замена
    ' =====================================

    Set xl = CreateObject("Excel.Application")

    Set book = xl.Workbooks.Open( _
        excelFile, 0, True)

    Set sh = book.Worksheets(1)

    last = sh.Cells( _
        sh.Rows.Count, 1).End(-4162).Row

    If last < 2 Then

        MsgBox "В Excel нет данных."
        GoTo CleanExit

    End If

    a = sh.Range("A2:D" & last).Value2

    book.Close False
    Set book = Nothing

    xl.Quit
    Set xl = Nothing


    ' =====================================
    ' РАЗБОР ОБОЗНАЧЕНИЙ И ПРИМЕЧАНИЙ
    ' =====================================

    total = 0
    token = 0

    For i = 1 To UBound(a, 1)

        If IsError(a(i, 1)) Or _
           IsError(a(i, 2)) Or _
           IsError(a(i, 4)) Then

            Err.Raise vbObjectError + 100, , _
                "Ошибка Excel в строке " & _
                CStr(i + 1)

        End If

        If IsEmpty(a(i, 1)) Then
            GoTo NextExcelRow
        End If

        ref = Trim(CStr(a(i, 1)))

        If ref = "" Then
            GoTo NextExcelRow
        End If

        If IsEmpty(a(i, 2)) Then

            Err.Raise vbObjectError + 101, , _
                "Нет наименования в строке " & _
                CStr(i + 1)

        End If

        SplitName CStr(a(i, 2)), model, maker

        replacement = ""

        If Not IsEmpty(a(i, 4)) Then
            replacement = Trim(CStr(a(i, 4)))
        End If

        ref = Replace(ref, "–", "-")
        ref = Replace(ref, "—", "-")
        ref = Replace(ref, ",", " ")
        ref = Replace(ref, ";", " ")
        ref = Replace(ref, vbTab, " ")
        ref = Replace(ref, Chr(160), " ")

        Do While InStr(ref, "  ") > 0
            ref = Replace(ref, "  ", " ")
        Loop

        parts = Split(ref, " ")

        For Each part In parts

            ref = Trim(CStr(part))

            If ref <> "" Then

                pos = InStr(ref, "-")

                If pos > 0 Then

                    firstRef = Left(ref, pos - 1)
                    lastRef = Mid(ref, pos + 1)

                Else

                    firstRef = ref
                    lastRef = ref

                End If

                code = RefCode(firstRef)
                endCode = RefCode(lastRef)

                ' Исключаем KT и КТ

                If IsExcludedCode(code) Then
                    GoTo NextReference
                End If

                n1 = RefNum(firstRef)
                n2 = RefNum(lastRef)

                If code = "" Or n1 < 1 Or _
                   n2 < n1 Or _
                   (endCode <> "" And _
                    endCode <> code) Then

                    Err.Raise vbObjectError + 102, , _
                        "Ошибка обозначения: " & ref

                End If

                ' =================================
                ' СОСТАВЛЯЕМ ТЕКСТ ПРИМЕЧАНИЯ
                ' =================================

                noteText = ""

                If replacement <> "" Then

                    noteText = BuildNoteText( _
                        code, model, maker, replacement)

                End If

                token = token + 1

                For num = n1 To n2

                    total = total + 1
                    ReDim Preserve el(1 To total)

                    el(total).Code = code
                    el(total).Num = num

                    el(total).Model = model
                    el(total).Maker = maker

                    el(total).TokenId = token
                    el(total).NoteText = noteText

                Next num

            End If

NextReference:

        Next part

NextExcelRow:

    Next i

    If total = 0 Then

        MsgBox "Нет элементов для перечня."
        GoTo CleanExit

    End If


    ' =====================================
    ' СОРТИРОВКА
    ' =====================================

    For i = 1 To total - 1

        For j = i + 1 To total

            If el(j).Code < el(i).Code Or _
               (el(j).Code = el(i).Code And _
                el(j).Num < el(i).Num) Then

                tmp = el(i)
                el(i) = el(j)
                el(j) = tmp

            End If

        Next j

    Next i


    ' =====================================
    ' ПРОВЕРКА ПОВТОРОВ
    ' =====================================

    For i = 2 To total

        If el(i).Code = el(i - 1).Code And _
           el(i).Num = el(i - 1).Num Then

            Err.Raise vbObjectError + 103, , _
                "Повтор обозначения: " & _
                el(i).Code & el(i).Num

        End If

    Next i


    ' =====================================
    ' ОБЪЕДИНЕНИЕ ЭЛЕМЕНТОВ
    '
    ' Теперь учитывается и примечание.
    ' Разные замены не объединяются.
    ' =====================================

    recCount = 0
    i = 1

    Do While i <= total

        j = i

        Do While j < total

            If el(j + 1).Code <> el(i).Code Then
                Exit Do
            End If

            If el(j + 1).Num <> el(j).Num + 1 Then
                Exit Do
            End If

            If StrComp( _
                el(j + 1).Model, el(i).Model, _
                vbTextCompare) <> 0 Then

                Exit Do

            End If

            If StrComp( _
                el(j + 1).Maker, el(i).Maker, _
                vbTextCompare) <> 0 Then

                Exit Do

            End If

            If StrComp( _
                el(j + 1).NoteText, _
                el(i).NoteText, _
                vbTextCompare) <> 0 Then

                Exit Do

            End If

            j = j + 1

        Loop

        recCount = recCount + 1
        ReDim Preserve rec(1 To recCount)

        rec(recCount).Ref = BuildReference(el, i, j)
        rec(recCount).Code = el(i).Code

        rec(recCount).TypeKey = _
            TypeKey(el(i).Code, el(i).Model)

        rec(recCount).Group = _
            GroupName(el(i).Code, el(i).Model)

        rec(recCount).Model = el(i).Model
        rec(recCount).Maker = el(i).Maker
        rec(recCount).Qty = j - i + 1

        rec(recCount).NoteText = el(i).NoteText

        i = j + 1

    Loop


    ' =====================================
    ' ПОИСК ОБЩИХ ТУ
    ' =====================================

    MarkRepeatedTU rec, recCount


    ' =====================================
    ' НУМЕРАЦИЯ УНИКАЛЬНЫХ ПРИМЕЧАНИЙ
    ' =====================================

    Set notes = New Collection

    NumberNotes rec, recCount, notes


    ' =====================================
    ' ПРОБНЫЕ ТАБЛИЦЫ
    ' =====================================

    Set testDoc1 = Documents.Add(Visible:=False)
    Set testDoc2 = Documents.Add(Visible:=False)

    SetupPEPage testDoc1.Sections(1)
    SetupPEPage testDoc2.Sections(1)

    Set test1 = testDoc1.Tables.Add( _
        testDoc1.Range(0, 0), 1, 4)

    Set test2 = testDoc2.Tables.Add( _
        testDoc2.Range(0, 0), 1, 4)

    FormatPETable test1, 1, True
    FormatPETable test2, 2, True


    ' =====================================
    ' ФОРМИРОВАНИЕ СТРАНИЦ КОМПОНЕНТОВ
    ' =====================================

    Set pages = New Collection
    Set pageRows = New Collection

    pages.Add pageRows

    limit = FIRST_ROWS

    ' Первая строка страницы пустая

    AddBlank pageRows

    i = 1

    Do While i <= recCount

        runEnd = i

        Do While runEnd < recCount

            If rec(runEnd + 1).TypeKey <> _
               rec(i).TypeKey Then

                Exit Do

            End If

            runEnd = runEnd + 1

        Loop


        ' =====================================
        ' ЗАПОЛНЕНИЕ ТЕКУЩЕГО ТИПА
        ' =====================================

        Do While i <= runEnd

            If pages.Count = 1 Then

                Set probe = test1

            Else

                Set probe = test2

            End If

            freeRows = limit - pageRows.Count

            maxFinish = i + freeRows - 1

            If maxFinish > runEnd Then
                maxFinish = runEnd
            End If

            found = False

            ' Ищем самый длинный фрагмент,
            ' помещающийся на странице

            For endIndex = maxFinish To i Step -1

                Set chunk = MakePageChunk( _
                    rec, i, endIndex, probe, _
                    pageRows.Count > 1)

                If chunk.Count <= freeRows Then

                    found = True
                    Exit For

                End If

            Next endIndex


            If Not found Then

                If pageRows.Count = 1 Then

                    Err.Raise vbObjectError + 104, , _
                        "Запись не помещается " & _
                        "на странице: " & rec(i).Ref

                End If

                NextPEPage pages, pageRows, limit

            Else

                For Each piece In chunk

                    pageRows.Add piece

                Next piece

                i = endIndex + 1

                If i <= runEnd Then

                    NextPEPage pages, pageRows, limit

                End If

            End If

        Loop

    Loop


    ' =====================================
    ' НОВАЯ ФУНКЦИЯ:
    ' СТРАНИЦЫ ПРИМЕЧАНИЙ
    ' =====================================

    If notes.Count > 0 Then

        AppendNotesToPages _
            pages, pageRows, limit, _
            notes, test2

    End If

    FillPage pageRows, limit

    testDoc1.Close wdDoNotSaveChanges
    testDoc2.Close wdDoNotSaveChanges

    Set testDoc1 = Nothing
    Set testDoc2 = Nothing


    ' =====================================
    ' ОТКРЫТИЕ WORD-ШАБЛОНА
    ' =====================================

    Set doc = Documents.Open( _
        FileName:=templateFile, _
        ReadOnly:=True, _
        AddToRecentFiles:=False)

    If doc.Sections.Count <> 3 Then

        Err.Raise vbObjectError + 105, , _
            "В шаблоне должно быть 3 раздела."

    End If

    baseName = Left(templateFile, _
        InStrRev(templateFile, ".") - 1)

    outputFile = baseName & "_PE.docx"

    k = 1

    Do While Dir(outputFile) <> ""

        outputFile = baseName & "_PE_" & _
                     CStr(k) & ".docx"

        k = k + 1

    Loop

    doc.SaveAs2 FileName:=outputFile, _
                FileFormat:=wdFormatXMLDocument


    ' =====================================
    ' ПОДГОТОВКА РАЗДЕЛОВ
    ' =====================================

    ClearSectionBody doc, 2
    ClearSectionBody doc, 1

    doc.Sections(2).PageSetup.SectionStart = _
        wdSectionNewPage

    doc.Sections(3).PageSetup.SectionStart = _
        wdSectionNewPage

    SetupPEPage doc.Sections(1)
    SetupPEPage doc.Sections(2)

    ' Третий раздел не изменяется


    ' =====================================
    ' СОЗДАНИЕ WORD-ТАБЛИЦ
    ' =====================================

    For k = 1 To pages.Count

        Set pageRows = pages(k)

        WritePEPage doc, pageRows, k

    Next k

    doc.Repaginate
    doc.Save

    MsgBox "Перечень готов!" & vbCrLf & _
           "Страниц перечня: " & _
           CStr(pages.Count) & vbCrLf & _
           "Уникальных примечаний: " & _
           CStr(notes.Count) & vbCrLf & _
           "Файл: " & outputFile


CleanExit:

    On Error Resume Next

    If Not book Is Nothing Then
        book.Close False
    End If

    If Not xl Is Nothing Then
        xl.Quit
    End If

    If Not testDoc1 Is Nothing Then
        testDoc1.Close wdDoNotSaveChanges
    End If

    If Not testDoc2 Is Nothing Then
        testDoc2.Close wdDoNotSaveChanges
    End If

    Exit Sub


ErrorHandler:

    MsgBox "Ошибка " & Err.Number & _
           ": " & Err.Description

    On Error Resume Next

    If Not doc Is Nothing Then
        doc.Close wdDoNotSaveChanges
    End If

    Resume CleanExit

End Sub


' =====================================
' СОСТАВЛЕНИЕ ТЕКСТА ПРИМЕЧАНИЯ
' =====================================

Private Function BuildNoteText( _
    code As String, _
    model As String, _
    maker As String, _
    replacement As String) As String

    Dim originalName As String
    Dim newName As String

    Dim newModel As String
    Dim newMaker As String

    Dim prefix As String

    replacement = CleanSpaces(replacement)

    If replacement = "" Then Exit Function

    ' =====================================
    ' ЗАМЕНА ТОЛЬКО ТЕХНИЧЕСКИХ УСЛОВИЙ
    ' =====================================

    If IsTU(replacement) Then

        If Not IsTU(maker) Then

            Err.Raise vbObjectError + 120, , _
                "У компонента нет исходного ТУ " & _
                "для замены на " & replacement

        End If

        BuildNoteText = _
            "Допускается замена технических " & _
            "условий " & maker & _
            " на " & replacement

        Exit Function

    End If


    ' =====================================
    ' ЗАМЕНА КОМПОНЕНТА
    ' =====================================

    originalName = FullTypeName(model, code)

    If maker <> "" Then

        originalName = originalName & " " & maker

    End If

    SplitName replacement, newModel, newMaker

    ' Если тип заменяющего элемента
    ' не указан, добавляем тип исходного

    If Not HasExplicitType(newModel) Then

        prefix = PrefixName(code, model)

        If prefix <> "" Then

            newModel = prefix & " " & newModel

        End If

    End If

    newName = newModel

    If newMaker <> "" Then

        newName = newName & " " & newMaker

    End If

    ' Не создаём бессмысленную замену
    ' компонента самим собой

    If StrComp( _
        CleanSpaces(originalName), _
        CleanSpaces(newName), _
        vbTextCompare) = 0 Then

        Exit Function

    End If

    BuildNoteText = _
        "Допускается замена " & _
        SourceComponentForm(originalName) & _
        " на " & DestinationComponentForm(newName)

End Function


' =====================================
' ГРАММАТИЧЕСКАЯ ФОРМА ИСХОДНОГО ТИПА
'
' Конденсатор -> конденсатора
' Резистор -> резистора
' =====================================

Private Function SourceComponentForm( _
    s As String) As String

    s = Trim(s)

    If StartsWithTerm(s, "Конденсатор") Then

        SourceComponentForm = _
            "конденсатора" & Mid(s, 12)

    ElseIf StartsWithTerm(s, "Резистор") Then

        SourceComponentForm = _
            "резистора" & Mid(s, 9)

    ElseIf StartsWithTerm(s, "Микросхема") Then

        SourceComponentForm = _
            "микросхемы" & Mid(s, 11)

    ElseIf StartsWithTerm(s, "Модуль питания") Then

        SourceComponentForm = _
            "модуля питания" & Mid(s, 15)

    ElseIf StartsWithTerm(s, "Генератор") Then

        SourceComponentForm = _
            "генератора" & Mid(s, 10)

    Else

        SourceComponentForm = _
            LowerFirst(s)

    End If

End Function


' =====================================
' ФОРМА ЗАМЕНЯЮЩЕГО КОМПОНЕНТА
' =====================================

Private Function DestinationComponentForm( _
    s As String) As String

    DestinationComponentForm = LowerFirst(s)

End Function


Private Function LowerFirst(s As String) As String

    s = Trim(s)

    If s = "" Then Exit Function

    LowerFirst = LCase(Left(s, 1)) & Mid(s, 2)

End Function


Private Function HasExplicitType( _
    s As String) As Boolean

    HasExplicitType = _
        StartsWithTerm(s, "Конденсатор") Or _
        StartsWithTerm(s, "Резистор") Or _
        StartsWithTerm(s, "Микросхема") Or _
        StartsWithTerm(s, "Модуль питания") Or _
        StartsWithTerm(s, "Генератор")

End Function


Private Function CleanSpaces( _
    s As String) As String

    s = Replace(s, Chr(160), " ")
    s = Replace(s, vbTab, " ")
    s = Trim(s)

    Do While InStr(s, "  ") > 0

        s = Replace(s, "  ", " ")

    Loop

    CleanSpaces = s

End Function


' =====================================
' НУМЕРАЦИЯ УНИКАЛЬНЫХ ПРИМЕЧАНИЙ
' =====================================

Private Sub NumberNotes( _
    rec() As PERecord, _
    recCount As Long, _
    notes As Collection)

    Dim i As Long
    Dim j As Long
    Dim noteNumber As Long

    Dim s As String

    For i = 1 To recCount

        s = CleanSpaces(rec(i).NoteText)

        If s <> "" Then

            noteNumber = 0

            ' Ищем уже существующее
            ' примечание с тем же текстом

            For j = 1 To notes.Count

                If StrComp( _
                    CStr(notes(j)), s, _
                    vbTextCompare) = 0 Then

                    noteNumber = j
                    Exit For

                End If

            Next j

            If noteNumber = 0 Then

                notes.Add s
                noteNumber = notes.Count

            End If

            rec(i).NoteNo = noteNumber

        End If

    Next i

End Sub


' =====================================
' ДОБАВЛЕНИЕ ПРИМЕЧАНИЙ К ПЕРЕЧНЮ
'
' Примечания всегда находятся
' во втором разделе Word.
' =====================================

Private Sub AppendNotesToPages( _
    pages As Collection, _
    ByRef pageRows As Collection, _
    ByRef limit As Long, _
    notes As Collection, _
    t As Table)

    Dim rows As Collection
    Dim piece As Variant

    Dim i As Long
    Dim required As Long

    If notes.Count = 0 Then Exit Sub

    ' Если перечень закончился
    ' на первом листе, начинаем
    ' примечания во втором разделе

    If pages.Count = 1 Then

        NextPEPage pages, pageRows, limit

    End If

    ' =====================================
    ' ПРОВЕРКА МЕСТА ДЛЯ ЗАГОЛОВКА
    ' И ПЕРВОГО ПРИМЕЧАНИЯ
    ' =====================================

    Set rows = MakeNoteRows( _
        CStr(notes(1)), 1, t)

    required = rows.Count + 1

    ' Разделительная пустая строка
    ' после перечня компонентов

    If pageRows.Count > 1 Then

        required = required + 1

    End If

    If pageRows.Count + required > limit Then

        NextPEPage pages, pageRows, limit

    End If

    ' Разделитель, если выше уже
    ' есть компоненты

    If pageRows.Count > 1 Then

        AddBlank pageRows

    End If

    AddNotesTitle pageRows


    ' =====================================
    ' ЗАПИСЬ ВСЕХ ПРИМЕЧАНИЙ
    ' =====================================

    For i = 1 To notes.Count

        Set rows = MakeNoteRows( _
            CStr(notes(i)), i, t)

        ' Примечание переносится
        ' только целиком

        If pageRows.Count + rows.Count > limit Then

            NextPEPage pages, pageRows, limit

            ' Повторяем заголовок
            ' на новой странице примечаний

            AddNotesTitle pageRows

        End If

        If pageRows.Count + rows.Count > limit Then

            Err.Raise vbObjectError + 121, , _
                "Примечание " & CStr(i) & _
                " слишком длинное для страницы."

        End If

        For Each piece In rows

            pageRows.Add piece

        Next piece

    Next i

End Sub


' =====================================
' ЗАГОЛОВОК РАЗДЕЛА ПРИМЕЧАНИЙ
' =====================================

Private Sub AddNotesTitle( _
    rows As Collection)

    rows.Add Array( _
        "", "П р и м е ч а н и я:", _
        "", "", False, _
        wdAlignParagraphCenter, _
        False, False, "", True, False)

End Sub


' =====================================
' СТРОКИ ОДНОГО ПРИМЕЧАНИЯ
' =====================================

Private Function MakeNoteRows( _
    noteText As String, _
    noteNumber As Long, _
    t As Table) As Collection

    Dim result As New Collection
    Dim lines As Collection

    Dim i As Long
    Dim refText As String

    Set lines = WrapText(t, noteText, 2)

    For i = 1 To lines.Count

        refText = ""

        ' Номер только в первой строке

        If i = 1 Then

            refText = CStr(noteNumber) & "."

        End If

        result.Add Array( _
            refText, CStr(lines(i)), _
            "", "", False, _
            wdAlignParagraphCenter, _
            False, False, "", False, True)

    Next i

    Set MakeNoteRows = result

End Function


' =====================================
' СБОРКА ФРАГМЕНТА СТРАНИЦЫ
' =====================================

Private Function MakePageChunk( _
    rec() As PERecord, _
    first As Long, _
    last As Long, _
    t As Table, _
    needGap As Boolean) As Collection

    Dim result As New Collection
    Dim tuIds As New Collection

    Dim rows As Collection
    Dim piece As Variant
    Dim id As Variant

    Dim i As Long

    Dim hasHeader As Boolean
    Dim hideTU As Boolean

    ' Заголовок группы при наличии
    ' минимум двух записей на странице

    hasHeader = False

    If last > first Then

        If rec(first).Group <> "" Then

            hasHeader = True

        End If

    End If


    ' =====================================
    ' ВСЕ ОБЩИЕ ТУ НА ЭТОЙ СТРАНИЦЕ
    ' =====================================

    If hasHeader Then

        For i = first To last

            If rec(i).TUBlockId > 0 Then

                If Not CollectionHasLong( _
                    tuIds, rec(i).TUBlockId) Then

                    tuIds.Add rec(i).TUBlockId

                End If

            End If

        Next i

    End If


    ' =====================================
    ' РАЗДЕЛИТЕЛЬ МЕЖДУ ТИПАМИ
    ' =====================================

    If needGap Then

        AddBlank result

    End If


    ' =====================================
    ' ЗАГОЛОВОК ГРУППЫ И ТУ
    ' =====================================

    If hasHeader Then

        result.Add Array( _
            "", rec(first).Group, _
            "", "", True, _
            wdAlignParagraphCenter, _
            False, False, "", False, False)

        For Each id In tuIds

            For i = first To last

                If rec(i).TUBlockId = CLng(id) Then

                    Set rows = MakeTUHeaderRows( _
                        rec(i), t)

                    AppendRows result, rows

                    Exit For

                End If

            Next i

        Next id

        ' Пустая строка после заголовка
        ' и всех общих ТУ

        AddBlank result

    End If


    ' =====================================
    ' КОМПОНЕНТЫ
    ' =====================================

    For i = first To last

        hideTU = False

        If hasHeader Then

            If rec(i).TUBlockId > 0 Then

                hideTU = CollectionHasLong( _
                    tuIds, rec(i).TUBlockId)

            End If

        End If

        Set rows = MakeItemRows( _
            rec(i), Not hasHeader, t, hideTU)

        AppendRows result, rows

    Next i

    Set MakePageChunk = result

End Function


Private Function CollectionHasLong( _
    values As Collection, _
    number As Long) As Boolean

    Dim item As Variant

    For Each item In values

        If CLng(item) = number Then

            CollectionHasLong = True
            Exit Function

        End If

    Next item

End Function

    destination As Collection, _
    source As Collection)

    Dim item As Variant

    For Each item In source

        destination.Add item

    Next item

End Sub

' =====================================
' ПОИСК ОБЩИХ ТУ ПО ВСЕЙ ГРУППЕ
' =====================================

Private Sub MarkRepeatedTU( _
    rec() As PERecord, _
    recCount As Long)

    Dim i As Long
    Dim j As Long
    Dim k As Long
    Dim p As Long

    Dim runStart As Long
    Dim runEnd As Long

    Dim blockId As Long
    Dim countTU As Long

    Dim series As String
    Dim otherSeries As String
    Dim maker As String

    blockId = 0
    runStart = 1

    Do While runStart <= recCount

        runEnd = runStart

        Do While runEnd < recCount

            If rec(runEnd + 1).TypeKey <> _
               rec(runStart).TypeKey Then

                Exit Do

            End If

            If rec(runEnd + 1).Group <> _
               rec(runStart).Group Then

                Exit Do

            End If

            runEnd = runEnd + 1

        Loop


        ' =====================================
        ' ИЩЕМ ПОВТОРЯЮЩИЕСЯ ТУ
        ' =====================================

        If rec(runStart).Group <> "" Then

            For i = runStart To runEnd

                If rec(i).TUBlockId = 0 And _
                   IsTU(rec(i).Maker) Then

                    maker = rec(i).Maker
                    countTU = 0

                    For j = runStart To runEnd

                        If StrComp( _
                            rec(j).Maker, maker, _
                            vbTextCompare) = 0 Then

                            countTU = countTU + 1

                        End If

                    Next j

                    If countTU >= MIN_TU_ITEMS Then

                        blockId = blockId + 1

                        series = ""

                        For j = runStart To runEnd

                            If StrComp( _
                                rec(j).Maker, maker, _
                                vbTextCompare) = 0 Then

                                series = SeriesCode( _
                                    rec(j).Model, _
                                    rec(j).Code)

                                Exit For

                            End If

                        Next j

                        For j = runStart To runEnd

                            If StrComp( _
                                rec(j).Maker, maker, _
                                vbTextCompare) = 0 Then

                                otherSeries = SeriesCode( _
                                    rec(j).Model, _
                                    rec(j).Code)

                                If StrComp( _
                                    series, otherSeries, _
                                    vbTextCompare) <> 0 Then

                                    series = ""
                                    Exit For

                                End If

                            End If

                        Next j

                        For p = runStart To runEnd

                            If StrComp( _
                                rec(p).Maker, maker, _
                                vbTextCompare) = 0 Then

                                rec(p).TUBlockId = blockId

                                rec(p).TUHeaderSeries = _
                                    series

                            End If

                        Next p

                    End If

                End If

            Next i

        End If

        runStart = runEnd + 1

    Loop

End Sub


' =====================================
' ПРОВЕРКА ТЕХНИЧЕСКИХ УСЛОВИЙ
' =====================================

Private Function IsTU(s As String) As Boolean

    s = UCase(Trim(s))

    If Len(s) < 3 Then Exit Function

    IsTU = (Right(s, 2) = "ТУ" And _
            InStr(s, ".") > 0)

End Function


' =====================================
' ОПРЕДЕЛЕНИЕ СЕРИИ КОМПОНЕНТА
' =====================================

Private Function SeriesCode( _
    model As String, _
    code As String) As String

    Dim s As String
    Dim word As String

    Dim p As Long
    Dim i As Long

    s = Trim(RemoveType(model, code))

    p = InStr(s, " ")

    If p <= 1 Then Exit Function

    word = Left(s, p - 1)

    If InStr(word, "-") = 0 Then
        Exit Function
    End If

    For i = 1 To Len(word)

        If Mid(word, i, 1) Like "#" Then

            SeriesCode = word
            Exit Function

        End If

    Next i

End Function


' =====================================
' СТРОКА ОБЩЕГО ТУ
' =====================================

Private Function MakeTUHeaderRows( _
    item As PERecord, _
    t As Table) As Collection

    Dim result As New Collection
    Dim lines As Collection

    Dim s As String
    Dim piece As Variant

    s = Trim(item.TUHeaderSeries)

    If s <> "" Then

        s = s & " "

    End If

    s = s & item.Maker

    Set lines = WrapText(t, s, 2)

    For Each piece In lines

        result.Add Array( _
            "", CStr(piece), "", "", _
            False, wdAlignParagraphCenter, _
            False, True, "", False, False)

    Next piece

    Set MakeTUHeaderRows = result

End Function


' =====================================
' ФИЗИЧЕСКИЕ СТРОКИ КОМПОНЕНТА
' =====================================

Private Function MakeItemRows( _
    item As PERecord, _
    fullType As Boolean, _
    t As Table, _
    Optional hideTU As Boolean = False) _
    As Collection

    Dim result As New Collection
    Dim chunks As New Collection

    Dim textLines As Collection
    Dim refLines As Collection
    Dim remarkLines As Collection

    Dim model As String
    Dim maker As String
    Dim remark As String

    Dim pair As Variant
    Dim piece As Variant

    Dim i As Long
    Dim n As Long
    Dim refAlign As Long

    Dim refText As String
    Dim qtyText As String
    Dim remarkText As String

    Dim hasRefWrap As Boolean

    model = item.Model
    maker = item.Maker

    ' =====================================
    ' ОБРАБОТКА ГРУППЫ
    ' =====================================

    If item.Group <> "" Then

        If fullType Then

            model = FullTypeName( _
                model, item.Code)

        Else

            model = RemoveType( _
                model, item.Code)

        End If

    End If

    ' Только ТУ убираем из модели,
    ' серия остаётся

    If hideTU Then

        maker = ""

    End If


    ' =====================================
    ' ОБОЗНАЧЕНИЕ И ПРИМЕЧАНИЕ
    ' =====================================

    Set refLines = WrapReference(t, item.Ref)

    hasRefWrap = (refLines.Count > 1)

    remark = ""

    If item.NoteNo > 0 Then

        remark = "См. примечание " & _
                 CStr(item.NoteNo)

    End If

    Set remarkLines = WrapText(t, remark, 4)


    ' =====================================
    ' МОДЕЛЬ И ИЗГОТОВИТЕЛЬ
    ' =====================================

    If hasRefWrap Then

        Set textLines = WrapText(t, model, 2)

        For Each piece In textLines

            chunks.Add Array(CStr(piece), "")

        Next piece

        Set textLines = WrapText(t, maker, 2)

        For Each piece In textLines

            chunks.Add Array("", CStr(piece))

        Next piece

    Else

        If model <> "" And maker <> "" And _
           TextFits( _
               t, model & Space(4) & maker, 2) Then

            chunks.Add Array(model, maker)

        Else

            Set textLines = WrapText(t, model, 2)

            For Each piece In textLines

                chunks.Add Array(CStr(piece), "")

            Next piece

            Set textLines = WrapText(t, maker, 2)

            For Each piece In textLines

                chunks.Add Array("", CStr(piece))

            Next piece

        End If

    End If

    If chunks.Count = 0 Then

        chunks.Add Array("", "")

    End If


    ' =====================================
    ' ЧИСЛО ФИЗИЧЕСКИХ СТРОК
    ' =====================================

    n = chunks.Count

    If refLines.Count > n Then
        n = refLines.Count
    End If

    If remarkLines.Count > n Then
        n = remarkLines.Count
    End If


    ' =====================================
    ' ЗАПОЛНЕНИЕ СТРОК
    ' =====================================

    For i = 1 To n

        If i <= chunks.Count Then

            pair = chunks(i)

        Else

            pair = Array("", "")

        End If

        refText = ""
        qtyText = ""
        remarkText = ""

        If i <= refLines.Count Then

            refText = CStr(refLines(i))

        End If

        If i <= remarkLines.Count Then

            remarkText = CStr(remarkLines(i))

        End If

        refAlign = wdAlignParagraphCenter

        If hasRefWrap Then

            If i = 1 Then

                refAlign = wdAlignParagraphLeft

            ElseIf i = refLines.Count Then

                refAlign = wdAlignParagraphRight

            End If

        End If

        If i = n Then

            qtyText = CStr(item.Qty)

        End If

        result.Add Array( _
            refText, _
            CStr(pair(0)), _
            CStr(pair(1)), _
            qtyText, _
            False, _
            refAlign, _
            (hasRefWrap And refText <> ""), _
            False, _
            remarkText, _
            False, _
            False)

    Next i

    Set MakeItemRows = result

End Function

' =====================================
' СОСТАВЛЕНИЕ ОБОЗНАЧЕНИЯ
' =====================================

Private Function BuildReference( _
    el() As PEItem, _
    first As Long, _
    last As Long) As String

    Dim k As Long
    Dim j As Long

    Dim result As String
    Dim part As String

    result = ""
    k = first

    Do While k <= last

        j = k

        Do While j < last

            If el(j + 1).TokenId <> _
               el(k).TokenId Then

                Exit Do

            End If

            If el(j + 1).Num <> _
               el(j).Num + 1 Then

                Exit Do

            End If

            j = j + 1

        Loop

        If j = k Then

            part = el(k).Code & el(k).Num

        Else

            part = el(k).Code & el(k).Num & _
                   "-" & el(j).Code & el(j).Num

        End If

        If result <> "" Then

            result = result & ","

        End If

        result = result & part

        k = j + 1

    Loop

    BuildReference = result

End Function


' =====================================
' КЛЮЧ ТИПА ЭЛЕМЕНТА
' =====================================

Private Function TypeKey( _
    code As String, _
    model As String) As String

    Select Case code

        Case "D", "DA", "DD", "U"

            TypeKey = "MICRO"

        Case "G"

            Select Case GComponentKind(model)

                Case "POWER"

                    TypeKey = "G_POWER"

                Case "GENERATOR"

                    TypeKey = "G_GENERATOR"

                Case Else

                    TypeKey = "G_OTHER"

            End Select

        Case Else

            TypeKey = code

    End Select

End Function


' =====================================
' КЛАССИФИКАЦИЯ G
' =====================================

Private Function GComponentKind( _
    model As String) As String

    If StartsWithTerm( _
        model, "Модуль питания") Then

        GComponentKind = "POWER"

    ElseIf StartsWithTerm( _
        model, "Генератор") Then

        GComponentKind = "GENERATOR"

    Else

        GComponentKind = "OTHER"

    End If

End Function


' =====================================
' ПРОВЕРКА НАЧАЛА НАИМЕНОВАНИЯ
' =====================================

Private Function StartsWithTerm( _
    s As String, _
    term As String) As Boolean

    Dim nextChar As String

    s = Trim(s)

    If term = "" Then Exit Function
    If Len(s) < Len(term) Then Exit Function

    If StrComp( _
        Left(s, Len(term)), _
        term, vbTextCompare) <> 0 Then

        Exit Function

    End If

    If Len(s) = Len(term) Then

        StartsWithTerm = True
        Exit Function

    End If

    nextChar = Mid(s, Len(term) + 1, 1)

    StartsWithTerm = _
        (nextChar = " " Or _
         nextChar = "-" Or _
         nextChar = "," Or _
         nextChar = "(")

End Function


' =====================================
' СОЗДАНИЕ WORD-ТАБЛИЦЫ
' =====================================

Private Sub WritePEPage( _
    doc As Document, _
    rows As Collection, _
    pageNo As Long)

    Dim rg As Range
    Dim t As Table
    Dim data As Variant

    Dim r As Long
    Dim pos As Long
    Dim sectionNum As Long

    If pageNo = 1 Then

        sectionNum = 1

    Else

        sectionNum = 2

    End If

    If pageNo <= 2 Then

        pos = doc.Sections(sectionNum).Range.Start

        Set rg = doc.Range(pos, pos)

    Else

        pos = doc.Sections(2).Range.End - 1

        Set rg = doc.Range(pos, pos)

        rg.InsertBreak wdPageBreak

        pos = doc.Sections(2).Range.End - 1

        Set rg = doc.Range(pos, pos)

    End If

    Set t = doc.Tables.Add( _
        rg, rows.Count, 4)

    FormatPETable t, pageNo

    For r = 1 To rows.Count

        data = rows(r)

        WritePERow t, r, data

    Next r

End Sub

    End If

' =====================================
' ФОРМАТИРОВАНИЕ ТАБЛИЦЫ
' =====================================

Private Sub FormatPETable( _
    t As Table, _
    Optional pageNo As Long = 1, _
    Optional forMeasure As Boolean = False)

    Dim c As Cell

    t.AutoFitBehavior wdAutoFitFixed


    ' =====================================
    ' ШИРИНЫ СТОЛБЦОВ
    ' =====================================

    If pageNo = 1 Then

        t.Columns(1).Width = _
            CentimetersToPoints(1.99)

        t.Columns(2).Width = _
            CentimetersToPoints(11.02)

        t.Columns(3).Width = _
            CentimetersToPoints(1)

        t.Columns(4).Width = _
            CentimetersToPoints(4.2)

    Else

        t.Columns(1).Width = _
            CentimetersToPoints(2)

        t.Columns(2).Width = _
            CentimetersToPoints(11)

        t.Columns(3).Width = _
            CentimetersToPoints(1)

        t.Columns(4).Width = _
            CentimetersToPoints(4.5)

    End If


    ' =====================================
    ' ГРАНИЦЫ И ОТСТУПЫ
    ' =====================================

    With t

        .Borders.Enable = True

        .TopPadding = 0
        .BottomPadding = 0

        .LeftPadding = _
            CentimetersToPoints(0.1)

        .RightPadding = _
            CentimetersToPoints(0.2)

    End With

    For Each c In t.Columns(1).Cells

        If forMeasure Then

            c.LeftPadding = _
                CentimetersToPoints(REF_PADDING)

            c.RightPadding = _
                CentimetersToPoints(REF_PADDING)

        Else

            c.LeftPadding = 0
            c.RightPadding = 0

        End If

    Next c


    ' =====================================
    ' ВЫСОТА И ПОЛОЖЕНИЕ ТАБЛИЦЫ
    ' =====================================

    With t.Rows

        .AllowBreakAcrossPages = False

        If forMeasure Then

            .HeightRule = wdRowHeightAtLeast

        Else

            .HeightRule = wdRowHeightExactly

            .Height = CentimetersToPoints(0.8)

            .WrapAroundText = True

            .RelativeHorizontalPosition = _
                wdRelativeHorizontalPositionPage

            .RelativeVerticalPosition = _
                wdRelativeVerticalPositionPage

            If pageNo = 1 Then

                .HorizontalPosition = _
                    CentimetersToPoints(2.12)

                .VerticalPosition = _
                    CentimetersToPoints(2.13)

            Else

                .HorizontalPosition = _
                    CentimetersToPoints(1.84)

                .VerticalPosition = _
                    CentimetersToPoints(2.38)

            End If

            .AllowOverlap = False

        End If

    End With


    ' =====================================
    ' ШРИФТ И АБЗАЦЫ
    ' =====================================

    t.Range.Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

    With t.Range

        .Font.Name = "Arial"
        .Font.Size = 12

        With .ParagraphFormat

            .SpaceBefore = 0
            .SpaceAfter = 0

            .LeftIndent = 0
            .RightIndent = 0
            .FirstLineIndent = 0

            .LineSpacingRule = wdLineSpaceSingle

            .KeepWithNext = False
            .KeepTogether = False
            .WidowControl = False

        End With

    End With

End Sub


' =====================================
' ЗАПОЛНЕНИЕ ФИЗИЧЕСКОЙ СТРОКИ
'
' data(0) - позиционное обозначение
' data(1) - наименование
' data(2) - изготовитель / ТУ
' data(3) - количество
' data(4) - заголовок группы
' data(5) - выравнивание обозначения
' data(6) - перенос обозначения
' data(7) - строка общего ТУ
' data(8) - ссылка на примечание
' data(9) - заголовок примечаний
' data(10) - текст примечания
' =====================================

Private Sub WritePERow( _
    t As Table, _
    rowNum As Long, _
    data As Variant)

    Dim model As String
    Dim maker As String
    Dim remark As String

    Dim isHeader As Boolean
    Dim isTUHeader As Boolean
    Dim isNotesTitle As Boolean
    Dim isNoteEntry As Boolean

    Dim refAlign As Long
    Dim refWrapped As Boolean

    Dim tabPos As Single
    Dim c As Cell

    model = CStr(data(1))
    maker = CStr(data(2))
    remark = CStr(data(8))

    isHeader = CBool(data(4))
    refAlign = CLng(data(5))
    refWrapped = CBool(data(6))
    isTUHeader = CBool(data(7))
    isNotesTitle = CBool(data(9))
    isNoteEntry = CBool(data(10))


    ' =====================================
    ' ПЕРВАЯ ГРАФА
    ' =====================================

    Set c = t.Cell(rowNum, 1)

    If refWrapped Then

        c.LeftPadding = _
            CentimetersToPoints(REF_PADDING)

        c.RightPadding = _
            CentimetersToPoints(REF_PADDING)

    Else

        c.LeftPadding = 0
        c.RightPadding = 0

    End If

    With c.Range

        .Text = CStr(data(0))

        .ParagraphFormat.Alignment = refAlign

    End With


    ' =====================================
    ' ТРЕТЬЯ ГРАФА: КОЛИЧЕСТВО
    ' =====================================

    With t.Cell(rowNum, 3).Range

        .Text = CStr(data(3))

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With


    ' =====================================
    ' ЧЕТВЁРТАЯ ГРАФА: ПРИМЕЧАНИЕ
    ' =====================================

    With t.Cell(rowNum, 4).Range

        .Text = remark

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With


    ' =====================================
    ' ВТОРАЯ ГРАФА: НАИМЕНОВАНИЕ
    ' =====================================

    Set c = t.Cell(rowNum, 2)

    tabPos = t.Columns(2).Width - _
             CentimetersToPoints(0.3)

    If isTUHeader Then

        c.LeftPadding = 0
        c.RightPadding = 0

    End If

    With c.Range

        If isHeader Or isTUHeader Then

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphCenter

        ElseIf isNotesTitle Then

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        ElseIf isNoteEntry Then

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        ElseIf model <> "" And maker <> "" Then

            .Text = model & vbTab & maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

            With .ParagraphFormat.TabStops

                .ClearAll

                .Add Position:=tabPos, _
                     Alignment:=wdAlignTabRight

            End With

        ElseIf maker <> "" Then

            .Text = maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphRight

        Else

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        End If

    End With

    t.Rows(rowNum).Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

End Sub


' =====================================
' ПРОВЕРКА ШИРИНЫ ТЕКСТА
' =====================================

Private Function TextFits( _
    t As Table, _
    s As String, _
    column As Long) As Boolean

    Dim rg As Range
    Dim lineCount As Long

    If s = "" Then

        TextFits = True
        Exit Function

    End If

    t.Cell(1, column).Range.Text = s

    Set rg = t.Cell(1, column).Range.Duplicate

    rg.Font.Name = "Arial"
    rg.Font.Size = 12

    rg.End = rg.End - 2

    rg.Document.Repaginate

    lineCount = rg.ComputeStatistics(wdStatisticLines)

    TextFits = (lineCount = 1)

End Function


' =====================================
' ПЕРЕНОС ОБЫЧНОГО ТЕКСТА
' =====================================

Private Function WrapText( _
    t As Table, _
    s As String, _
    column As Long) As Collection

    Dim result As New Collection

    Dim words As Variant
    Dim word As Variant

    Dim line As String
    Dim candidate As String
    Dim part As String
    Dim n As Long

    s = CleanSpaces(s)

    If s = "" Then

        Set WrapText = result
        Exit Function

    End If

    words = Split(s, " ")

    For Each word In words

        part = CStr(word)

        If line = "" Then

            candidate = part

        Else

            candidate = line & " " & part

        End If

        If TextFits(t, candidate, column) Then

            line = candidate

        Else

            If line <> "" Then

                result.Add line
                line = ""

            End If

            If TextFits(t, part, column) Then

                line = part

            Else

                ' Длинное слово делится
                ' на физические строки

                Do While part <> ""

                    n = LongestPart(t, part, column)

                    If n < 1 Then

                        Err.Raise _
                            vbObjectError + 130, , _
                            "Не помещается символ: " & _
                            Left(part, 1)

                    End If

                    result.Add Left(part, n)

                    part = Mid(part, n + 1)

                Loop

            End If

        End If

    Next word

    If line <> "" Then

        result.Add line

    End If

    Set WrapText = result

End Function


' =====================================
' ПЕРЕНОС ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
' =====================================

Private Function WrapReference( _
    t As Table, _
    s As String) As Collection

    Dim result As New Collection
    Dim items As Variant

    Dim item As String
    Dim suffix As String
    Dim line As String
    Dim candidate As String

    Dim leftPart As String
    Dim rightPart As String

    Dim i As Long
    Dim p As Long

    Dim firstNum As Long
    Dim lastNum As Long
    Dim previousNum As Long

    Dim forceRange As Boolean
    Dim forceNewLine As Boolean

    If s = "" Then

        Set WrapReference = result
        Exit Function

    End If

    items = Split(s, ",")

    previousNum = 0
    line = ""

    For i = LBound(items) To UBound(items)

        item = Trim(CStr(items(i)))
        suffix = ""

        If i < UBound(items) Then

            suffix = ","

        End If

        p = InStr(item, "-")

        If p > 0 Then

            leftPart = Left(item, p)
            rightPart = Mid(item, p + 1)

            firstNum = RefNum( _
                Left(item, p - 1))

            lastNum = RefNum(rightPart)

        Else

            firstNum = RefNum(item)
            lastNum = firstNum

        End If

        forceRange = False

        If p > 0 Then

            If firstNum > 100 Or _
               lastNum > 100 Then

                forceRange = True

            End If

        End If


        ' =====================================
        ' ПЕРЕНОС ДИАПАЗОНА
        ' =====================================

        If p > 0 And _
           (forceRange Or _
            Not TextFits( _
                t, item & suffix, 1)) Then

            If line <> "" Then

                result.Add line
                line = ""

            End If

            If Not TextFits( _
                t, leftPart, 1) Then

                Err.Raise vbObjectError + 131, , _
                    "Не помещается: " & leftPart

            End If

            result.Add leftPart

            candidate = rightPart & suffix

            If Not TextFits( _
                t, candidate, 1) Then

                Err.Raise vbObjectError + 132, , _
                    "Не помещается: " & candidate

            End If

            line = candidate

        Else

            candidate = item & suffix
            forceNewLine = False

            If line <> "" Then

                If previousNum > 100 Or _
                   firstNum > 100 Then

                    forceNewLine = True

                End If

            End If

            If line = "" Then

                If Not TextFits( _
                    t, candidate, 1) Then

                    Err.Raise vbObjectError + 133, , _
                        "Не помещается: " & candidate

                End If

                line = candidate

            ElseIf Not forceNewLine And _
                   TextFits( _
                       t, line & candidate, 1) Then

                line = line & candidate

            Else

                result.Add line

                If Not TextFits( _
                    t, candidate, 1) Then

                    Err.Raise vbObjectError + 134, , _
                        "Не помещается: " & candidate

                End If

                line = candidate

            End If

        End If

        previousNum = lastNum

    Next i

    If line <> "" Then

        result.Add line

    End If

    Set WrapReference = result

End Function


' =====================================
' МАКСИМАЛЬНАЯ ДЛИНА ФРАГМЕНТА
' =====================================

Private Function LongestPart( _
    t As Table, _
    s As String, _
    column As Long) As Long

    Dim lo As Long
    Dim hi As Long
    Dim mid As Long

    If Len(s) = 0 Then Exit Function

    If Not TextFits( _
        t, Left(s, 1), column) Then

        LongestPart = 0
        Exit Function

    End If

    lo = 1
    hi = Len(s)

    Do While lo < hi

        mid = (lo + hi + 1) \ 2

        If TextFits( _
            t, Left(s, mid), column) Then

            lo = mid

        Else

            hi = mid - 1

        End If

    Loop

    LongestPart = lo

End Function


' =====================================
' РАЗДЕЛЕНИЕ МОДЕЛИ И ФИРМЫ / ТУ
' =====================================

Private Sub SplitName( _
    s As String, _
    ByRef model As String, _
    ByRef maker As String)

    Dim p As Long
    Dim word As String

    s = CleanSpaces(s)

    model = s
    maker = ""

    p = InStr( _
        1, s, " ф. ", vbTextCompare)

    If p > 0 Then

        model = Trim(Left(s, p - 1))
        maker = Trim(Mid(s, p + 1))

    Else

        p = InStrRev(s, " ")

        If p > 0 Then

            word = Trim(Mid(s, p + 1))

            If IsTU(word) Then

                model = Trim(Left(s, p - 1))
                maker = word

            End If

        End If

    End If

    model = NormalizeModelName(model)

End Sub


' =====================================
' НОРМАЛИЗАЦИЯ НАИМЕНОВАНИЙ
' =====================================

Private Function NormalizeModelName( _
    s As String) As String

    s = Trim(s)

    s = ReplaceLeading(s, _
        "Модуль электропитания", _
        "Модуль питания")

    s = ReplaceLeading(s, _
        "Источник электропитания", _
        "Модуль питания")

    s = ReplaceLeading(s, _
        "Источник питания", _
        "Модуль питания")

    ' Новые замены добавлять здесь:
    '
    ' s = ReplaceLeading(s, _
    '     "Старое название", _
    '     "Новое название")

    NormalizeModelName = s

End Function


' =====================================
' ЗАМЕНА НАЧАЛА НАИМЕНОВАНИЯ
' =====================================

Private Function ReplaceLeading( _
    s As String, _
    oldText As String, _
    newText As String) As String

    s = Trim(s)

    If StartsWithTerm(s, oldText) Then

        ReplaceLeading = newText & _
            Mid(s, Len(oldText) + 1)

    Else

        ReplaceLeading = s

    End If

End Function


' =====================================
' БУКВЕННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Private Function RefCode( _
    s As String) As String

    Dim i As Long

    s = UCase(Trim(s))

    If Len(s) = 0 Then Exit Function

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            Exit For

        End If

    Next i

    If i <= Len(s) Then

        RefCode = Left(s, i - 1)

    Else

        RefCode = ""

    End If

End Function


' =====================================
' ЧИСЛОВОЕ ОБОЗНАЧЕНИЕ
' =====================================

Private Function RefNum( _
    s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function


' =====================================
' ПОЛНОЕ НАИМЕНОВАНИЕ ТИПА
' =====================================

Private Function PrefixName( _
    code As String, _
    model As String) As String

    Select Case code

        Case "C"

            PrefixName = "Конденсатор"

        Case "R"

            PrefixName = "Резистор"

        Case "D", "DA", "DD", "U"

            PrefixName = "Микросхема"

        Case "G"

            If GComponentKind(model) = "POWER" Then

                PrefixName = "Модуль питания"

            Else

                PrefixName = ""

            End If

        Case Else

            PrefixName = ""

    End Select

End Function


' =====================================
' УДАЛЕНИЕ ТИПА ИЗ НАЗВАНИЯ
' =====================================

Private Function RemoveType( _
    s As String, _
    code As String) As String

    Dim prefix As String

    s = Trim(s)
    prefix = PrefixName(code, s)

    If prefix <> "" Then

        If StartsWithTerm(s, prefix) Then

            s = Trim(Mid(s, Len(prefix) + 1))

        End If

    End If

    RemoveType = s

End Function


' =====================================
' ПОЛНОЕ НАЗВАНИЕ БЕЗ ЗАГОЛОВКА
' =====================================

Private Function FullTypeName( _
    s As String, _
    code As String) As String

    Dim prefix As String

    s = Trim(s)
    prefix = PrefixName(code, s)

    If prefix = "" Then

        FullTypeName = s
        Exit Function

    End If

    If StartsWithTerm( _
        s, "Модуль питания") Or _
       StartsWithTerm( _
        s, "Генератор") Then

        FullTypeName = s
        Exit Function

    End If

    If StartsWithTerm(s, prefix) Then

        FullTypeName = s

    Else

        FullTypeName = prefix & " " & s

    End If

End Function

End Function
