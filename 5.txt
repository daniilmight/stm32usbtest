
Option Explicit

Private Type PEItem
    Code As String
    Num As Long
    Model As String
    Maker As String
End Type

Private Const FIRST_ROWS As Long = 23
Private Const NEXT_ROWS As Long = 30

' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Sub MakeWordPE()

    Dim xl As Object, book As Object, sh As Object
    Dim doc As Document, testDoc As Document
    Dim testTable As Table, t As Table
    Dim rg As Range

    Dim a As Variant
    Dim el() As PEItem, tmp As PEItem
    Dim parts As Variant, part As Variant
    Dim pair As Variant, rowData As Variant
    Dim piece As Variant

    Dim pages As Collection
    Dim pageRows As Collection
    Dim chunks As Collection
    Dim refLines As Collection
    Dim textLines As Collection

    Dim templateFile As String
    Dim excelFile As String
    Dim outputFile As String
    Dim baseName As String

    Dim ref As String, code As String
    Dim endCode As String
    Dim firstRef As String, lastRef As String
    Dim model As String, maker As String
    Dim grp As String, oldGrp As String
    Dim refText As String, qtyText As String

    Dim last As Long, total As Long
    Dim i As Long, j As Long, k As Long
    Dim num As Long, n1 As Long, n2 As Long
    Dim pos As Long, n As Long
    Dim need As Long, limit As Long

    Dim pg As Long, r As Long
    Dim baseRow As Long, rowsCount As Long

    On Error GoTo ErrorHandler

    ' =====================================
    ' ИЗМЕНЕНИЕ 1
    ' ВЫБОР ШАБЛОНА WORD
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите шаблон Word"
        .Filters.Clear
        .Filters.Add "Word", "*.doc;*.docx;*.docm"

        If .Show <> -1 Then Exit Sub
        templateFile = .SelectedItems(1)

    End With

    ' Выбор Excel

    With Application.FileDialog(3)

        .Title = "Выберите перечень Excel"
        .Filters.Clear
        .Filters.Add "Excel", "*.xlsx;*.xlsm;*.xls"

        If .Show <> -1 Then Exit Sub
        excelFile = .SelectedItems(1)

    End With

    ' Чтение Excel

    Set xl = CreateObject("Excel.Application")
    Set book = xl.Workbooks.Open(excelFile, 0, True)
    Set sh = book.Worksheets(1)

    last = sh.Cells(sh.Rows.Count, 1).End(-4162).Row

    If last < 2 Then
        MsgBox "В Excel нет данных."
        GoTo Finish
    End If

    a = sh.Range("A2:C" & last).Value2

    book.Close False
    Set book = Nothing

    xl.Quit
    Set xl = Nothing


    ' =====================================
    ' РАЗБОР ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
    ' =====================================

    total = 0

    For i = 1 To UBound(a, 1)

        SplitName CStr(a(i, 2)), model, maker

        ref = Trim(CStr(a(i, 1)))

        ref = Replace(ref, "–", "-")
        ref = Replace(ref, "—", "-")
        ref = Replace(ref, ",", " ")
        ref = Replace(ref, ";", " ")
        ref = Replace(ref, vbTab, " ")
        ref = Replace(ref, Chr(160), " ")

        parts = Split(ref, " ")

        For Each part In parts

            ref = Trim(CStr(part))

            If ref <> "" Then

                pos = InStr(ref, "-")

                If pos > 0 Then

                    firstRef = Left(ref, pos - 1)
                    lastRef = Mid(ref, pos + 1)

                Else

                    firstRef = ref
                    lastRef = ref

                End If

                code = RefCode(firstRef)
                endCode = RefCode(lastRef)

                n1 = RefNum(firstRef)
                n2 = RefNum(lastRef)

                If code = "" Or n1 < 1 Or _
                   n2 < n1 Or _
                   (endCode <> "" And endCode <> code) Then

                    Err.Raise vbObjectError + 100, , _
                        "Ошибка обозначения: " & ref

                End If

                For num = n1 To n2

                    total = total + 1

                    ReDim Preserve el(1 To total)

                    el(total).Code = code
                    el(total).Num = num
                    el(total).Model = model
                    el(total).Maker = maker

                Next num

            End If

        Next part

    Next i

    If total = 0 Then GoTo Finish


    ' =====================================
    ' СОРТИРОВКА
    ' =====================================

    For i = 1 To total - 1

        For j = i + 1 To total

            If el(j).Code < el(i).Code Or _
               (el(j).Code = el(i).Code And _
                el(j).Num < el(i).Num) Then

                tmp = el(i)
                el(i) = el(j)
                el(j) = tmp

            End If

        Next j

    Next i

    ' Проверка повторов

    For i = 2 To total

        If el(i).Code = el(i - 1).Code And _
           el(i).Num = el(i - 1).Num Then

            Err.Raise vbObjectError + 101, , _
                "Повтор обозначения: " & _
                el(i).Code & el(i).Num

        End If

    Next i


    ' =====================================
    ' ВРЕМЕННАЯ ТАБЛИЦА ДЛЯ ИЗМЕРЕНИЙ
    ' =====================================

    Set testDoc = Documents.Add(Visible:=False)

    SetupPEPage testDoc.Sections(1)

    Set testTable = testDoc.Tables.Add( _
        testDoc.Range(0, 0), 1, 4)

    FormatPETable testTable

    ' Разрешаем увеличивать пробную строку

    testTable.Rows.HeightRule = wdRowHeightAtLeast


    ' =====================================
    ' ПОДГОТОВКА ДАННЫХ ПО СТРАНИЦАМ
    ' =====================================

    Set pages = New Collection
    Set pageRows = New Collection

    pages.Add pageRows

    limit = FIRST_ROWS
    oldGrp = ""

    i = 1

    Do While i <= total

        j = i

        ' Объединяем соседние одинаковые
        ' компоненты

        Do While j < total

            If el(j + 1).Code <> el(i).Code Then
                Exit Do
            End If

            If el(j + 1).Num <> el(j).Num + 1 Then
                Exit Do
            End If

            If el(j + 1).Model <> el(i).Model Or _
               el(j + 1).Maker <> el(i).Maker Then
                Exit Do
            End If

            j = j + 1

        Loop

        code = el(i).Code
        model = RemoveType(el(i).Model, code)
        maker = el(i).Maker

        If i = j Then

            ref = code & el(i).Num

        Else

            ref = code & el(i).Num & "-" & _
                  code & el(j).Num

        End If

        grp = GroupName(code)


        ' =====================================
        ' ПЕРЕНОС ТЕКСТА
        ' =====================================

        Set chunks = New Collection

        ' Проверяем модель и фирму вместе.
        ' Запас не менее трёх пробелов.

        If maker <> "" And _
           TextFits(testTable, _
                    model & Space(3) & maker, 2) Then

            chunks.Add Array(model, maker)

        Else

            Set textLines = WrapText( _
                testTable, model, 2)

            For Each piece In textLines

                chunks.Add Array(CStr(piece), "")

            Next piece

            Set textLines = WrapText( _
                testTable, maker, 2)

            For Each piece In textLines

                chunks.Add Array("", CStr(piece))

            Next piece

        End If

        If chunks.Count = 0 Then
            chunks.Add Array("", "")
        End If

        ' Обозначение также проверяем
        ' по фактической ширине первой графы

        Set refLines = WrapText(testTable, ref, 1)

        n = chunks.Count

        If refLines.Count > n Then
            n = refLines.Count
        End If

        ' =====================================
        ' РАСПРЕДЕЛЕНИЕ ПО СТРАНИЦАМ
        ' =====================================

        need = n

        If grp <> oldGrp Then
            need = need + 3
        End If

        If need > NEXT_ROWS Then

            Err.Raise vbObjectError + 102, , _
                "Слишком длинная запись: " & ref

        End If

        If pageRows.Count + need > limit Then

            FillPage pageRows, limit

            Set pageRows = New Collection
            pages.Add pageRows

            limit = NEXT_ROWS

        End If

        ' Групповой заголовок

        If grp <> oldGrp Then

            pageRows.Add Array("", "", "", "")

            pageRows.Add Array("", grp, "", "")

            pageRows.Add Array("", "", "", "")

            oldGrp = grp

        End If

        ' Каждая запись занимает физические
        ' строки таблицы

        For k = 1 To n

            If k <= chunks.Count Then

                pair = chunks(k)

            Else

                pair = Array("", "")

            End If

            refText = ""
            qtyText = ""

            If k <= refLines.Count Then
                refText = CStr(refLines(k))
            End If

            ' Количество только в конце

            If k = n Then
                qtyText = CStr(j - i + 1)
            End If

            pageRows.Add Array( _
                refText, _
                CStr(pair(0)), _
                CStr(pair(1)), _
                qtyText)

        Next k

        i = j + 1

    Loop

    FillPage pageRows, limit

    testDoc.Close wdDoNotSaveChanges
    Set testDoc = Nothing


    ' =====================================
    ' ИЗМЕНЕНИЕ 2
    ' ОТКРЫТИЕ ШАБЛОНА
    ' =====================================

    Set doc = Documents.Open( _
        FileName:=templateFile, _
        ReadOnly:=True, _
        AddToRecentFiles:=False)

    If doc.Sections.Count <> 3 Then

        Err.Raise vbObjectError + 103, , _
            "В шаблоне должно быть 3 раздела. " & _
            "Обнаружено: " & doc.Sections.Count

    End If

    ' Создание имени выходного документа

    baseName = Left(templateFile, _
        InStrRev(templateFile, ".") - 1)

    outputFile = baseName & "_PE.docx"

    k = 1

    Do While Dir(outputFile) <> ""

        outputFile = baseName & "_PE_" & k & ".docx"
        k = k + 1

    Loop

    ' Сохраняем отдельную копию

    doc.SaveAs2 FileName:=outputFile, _
                FileFormat:=wdFormatXMLDocument


    ' =====================================
    ' ИЗМЕНЕНИЕ 3
    ' СОХРАНЕНИЕ ТРЁХ РАЗДЕЛОВ
    ' =====================================

    ' Убираем старое содержимое второго
    ' раздела, включая лишние страницы.
    ' Разрыв раздела не удаляем.

    ClearSectionBody doc, 2

    ' Очищаем область первого листа

    ClearSectionBody doc, 1

    ' Третий раздел не трогаем!

    doc.Sections(2).PageSetup.SectionStart = _
        wdSectionNewPage

    doc.Sections(3).PageSetup.SectionStart = _
        wdSectionNewPage


    ' =====================================
    ' ИЗМЕНЕНИЕ 4
    ' КООРДИНАТЫ ТАБЛИЦЫ
    ' =====================================

    SetupPEPage doc.Sections(1)
    SetupPEPage doc.Sections(2)


    ' =====================================
    ' СОЗДАНИЕ ТАБЛИЦ ПЕРЕЧНЯ
    ' =====================================

    ' Первый раздел: первая страница

    Set rg = doc.Range( _
        doc.Sections(1).Range.Start, _
        doc.Sections(1).Range.Start)

    Set t = doc.Tables.Add(rg, FIRST_ROWS, 4)

    FormatPETable t

    Set pageRows = pages(1)

    For r = 1 To FIRST_ROWS

        rowData = pageRows(r)

        WritePERow t, r, rowData

    Next r


    ' =====================================
    ' ИЗМЕНЕНИЕ 5
    ' ВТОРОЙ РАЗДЕЛ - ВСЕ ОСТАЛЬНЫЕ ЛИСТЫ
    ' =====================================

    ' Как минимум одна страница
    ' второго раздела остаётся в документе.

    rowsCount = (pages.Count - 1) * NEXT_ROWS

    If rowsCount = 0 Then
        rowsCount = NEXT_ROWS
    End If

    Set rg = doc.Range( _
        doc.Sections(2).Range.Start, _
        doc.Sections(2).Range.Start)

    Set t = doc.Tables.Add(rg, rowsCount, 4)

    FormatPETable t

    For pg = 2 To pages.Count

        Set pageRows = pages(pg)

        baseRow = (pg - 2) * NEXT_ROWS

        ' Новый лист внутри второго раздела

        If pg > 2 Then

            t.Cell(baseRow + 1, 1).Range. _
                Paragraphs(1).PageBreakBefore = True

        End If

        For r = 1 To NEXT_ROWS

            rowData = pageRows(r)

            WritePERow t, baseRow + r, rowData

        Next r

    Next pg


    ' Третий раздел остаётся после таблицы

    doc.Repaginate
    doc.Save

    MsgBox "Перечень готов." & vbCrLf & _
           "Листов перечня: " & pages.Count & vbCrLf & _
           "Файл: " & outputFile

Finish:

    On Error Resume Next

    If Not book Is Nothing Then book.Close False
    If Not xl Is Nothing Then xl.Quit

    If Not testDoc Is Nothing Then
        testDoc.Close wdDoNotSaveChanges
    End If

    Exit Sub

ErrorHandler:

    MsgBox "Ошибка: " & Err.Description

    On Error Resume Next

    If Not doc Is Nothing Then
        doc.Close wdDoNotSaveChanges
    End If

    Resume Finish

End Sub


' =====================================
' ОЧИСТКА СОДЕРЖИМОГО РАЗДЕЛА
' БЕЗ УДАЛЕНИЯ РАЗРЫВА
' =====================================

Sub ClearSectionBody(doc As Document, _
                     sectionNum As Long)

    Dim rg As Range

    Set rg = doc.Sections(sectionNum).Range.Duplicate

    rg.End = rg.End - 1

    If rg.End > rg.Start Then
        rg.Delete
    End If

End Sub


' =====================================
' КООРДИНАТЫ И ПОЛЯ
' =====================================

Sub SetupPEPage(sec As Section)

    With sec.PageSetup

        .PaperSize = wdPaperA4
        .Orientation = wdOrientPortrait

        ' Таблица начинается:
        ' X = 2,12 см; Y = 2,13 см

        .LeftMargin = CentimetersToPoints(2.12)
        .TopMargin = CentimetersToPoints(2.13)

        ' Остальные поля

        .RightMargin = CentimetersToPoints(0.5)
        .BottomMargin = CentimetersToPoints(1)

    End With

End Sub


' =====================================
' РАЗМЕРЫ И ОФОРМЛЕНИЕ ТАБЛИЦЫ
' =====================================

Sub FormatPETable(t As Table)

    t.AutoFitBehavior wdAutoFitFixed

    ' Ширина граф, сантиметры

    t.Columns(1).Width = CentimetersToPoints(1.99)
    t.Columns(2).Width = CentimetersToPoints(11.02)
    t.Columns(3).Width = CentimetersToPoints(1)
    t.Columns(4).Width = CentimetersToPoints(4.2)

    t.Rows.SetLeftIndent _
        LeftIndent:=0, RulerStyle:=wdAdjustNone

    With t

        .Borders.Enable = True

        .TopPadding = 0
        .BottomPadding = 0

        ' Левый отступ 0,1 см
        .LeftPadding = CentimetersToPoints(0.1)

        ' ИЗМЕНЕНИЕ 6:
        ' Отступ справа для фирмы и ТУ
        .RightPadding = CentimetersToPoints(0.2)

    End With

    With t.Rows

        .HeightRule = wdRowHeightExactly
        .Height = CentimetersToPoints(0.8)

        .AllowBreakAcrossPages = False

    End With

    t.Range.Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

    With t.Range

        .Font.Name = "Arial"
        .Font.Size = 12

        With .ParagraphFormat

            .SpaceBefore = 0
            .SpaceAfter = 0

            .LeftIndent = 0
            .RightIndent = 0
            .FirstLineIndent = 0

            .LineSpacingRule = wdLineSpaceSingle

            .KeepWithNext = False
            .KeepTogether = False
            .WidowControl = False

        End With

    End With

End Sub


' =====================================
' ЗАПОЛНЕНИЕ ОДНОЙ СТРОКИ WORD
' =====================================

Sub WritePERow(t As Table, _
               rowNum As Long, _
               data As Variant)

    Dim model As String, maker As String

    model = CStr(data(1))
    maker = CStr(data(2))

    ' Позиционное обозначение

    With t.Cell(rowNum, 1).Range

        .Text = CStr(data(0))

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With

    ' Количество

    With t.Cell(rowNum, 3).Range

        .Text = CStr(data(3))

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With

    ' Наименование + изготовитель/ТУ

    With t.Cell(rowNum, 2).Range

        If model <> "" And maker <> "" Then

            ' Модель слева, фирма справа

            .Text = model & vbTab & maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

            With .ParagraphFormat.TabStops

                .ClearAll

                .Add Position:= _
                    CentimetersToPoints(10.72) - 1, _
                    Alignment:=wdAlignTabRight

            End With

        ElseIf maker <> "" Then

            ' Только фирма / ТУ справа

            .Text = maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphRight

        Else

            ' Только наименование

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        End If

    End With

    ' Выравнивание всех ячеек по высоте

    t.Rows(rowNum).Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

End Sub


' =====================================
' ДОПОЛНЕНИЕ СТРАНИЦЫ ПУСТЫМИ СТРОКАМИ
' =====================================

Sub FillPage(rows As Collection, _
             maxRows As Long)

    Do While rows.Count < maxRows

        rows.Add Array("", "", "", "")

    Loop

End Sub


' =====================================
' ПРОВЕРКА ШИРИНЫ ТЕКСТА
' =====================================

Function TextFits(t As Table, _
                  s As String, _
                  column As Long) As Boolean

    Dim rg As Range
    Dim count As Long

    If s = "" Then

        TextFits = True
        Exit Function

    End If

    t.Cell(1, column).Range.Text = s

    Set rg = t.Cell(1, column).Range

    rg.Font.Name = "Arial"
    rg.Font.Size = 12

    ' Убираем из измерения два
    ' служебных символа конца ячейки

    rg.End = rg.End - 2

    rg.Document.Repaginate

    count = rg.ComputeStatistics(wdStatisticLines)

    TextFits = (count <= 1)

End Function


' =====================================
' РАЗБИЕНИЕ ДЛИННОГО ТЕКСТА
' =====================================

Function WrapText(t As Table, _
                  s As String, _
                  column As Long) As Collection

    Dim result As New Collection
    Dim words As Variant, word As Variant

    Dim line As String
    Dim candidate As String
    Dim part As String
    Dim n As Long

    s = Trim(Replace(s, vbTab, " "))

    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop

    If s = "" Then
        Set WrapText = result
        Exit Function
    End If

    words = Split(s, " ")

    For Each word In words

        part = CStr(word)

        If line = "" Then
            candidate = part
        Else
            candidate = line & " " & part
        End If

        If TextFits(t, candidate, column) Then

            line = candidate

        Else

            If line <> "" Then
                result.Add line
                line = ""
            End If

            If TextFits(t, part, column) Then

                line = part

            Else

                ' Длинное слово делим на части

                Do While part <> ""

                    n = LongestPart(t, part, column)

                    result.Add Left(part, n)
                    part = Mid(part, n + 1)

                Loop

            End If

        End If

    Next word

    If line <> "" Then result.Add line

    Set WrapText = result

End Function


' =====================================
' ПОИСК ДЛИНЫ ПОДСТРОКИ
' =====================================

Function LongestPart(t As Table, _
                     s As String, _
                     column As Long) As Long

    Dim lo As Long, hi As Long, mid As Long

    lo = 1
    hi = Len(s)

    Do While lo < hi

        mid = (lo + hi + 1) \ 2

        If TextFits(t, Left(s, mid), column) Then

            lo = mid

        Else

            hi = mid - 1

        End If

    Loop

    LongestPart = lo

End Function


' =====================================
' РАЗДЕЛЕНИЕ МОДЕЛИ И ФИРМЫ/ТУ
' =====================================

Sub SplitName(s As String, _
              ByRef model As String, _
              ByRef maker As String)

    Dim p As Long, word As String

    s = Trim(s)

    model = s
    maker = ""

    p = InStr(1, s, " ф. ", vbTextCompare)

    If p > 0 Then

        model = Trim(Left(s, p - 1))
        maker = Trim(Mid(s, p + 1))

    Else

        p = InStrRev(s, " ")

        If p > 0 Then

            word = Trim(Mid(s, p + 1))

            If Len(word) > 2 And _
               Right(UCase(word), 2) = "ТУ" And _
               InStr(word, ".") > 0 Then

                model = Trim(Left(s, p - 1))
                maker = word

            End If

        End If

    End If

End Sub


' =====================================
' БУКВЕННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefCode(s As String) As String

    Dim i As Long

    s = UCase(Trim(s))

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then Exit For

    Next i

    RefCode = Left(s, i - 1)

End Function


' =====================================
' НОМЕР ЭЛЕМЕНТА
' =====================================

Function RefNum(s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function


' =====================================
' УДАЛЕНИЕ ПОВТОРЯЮЩЕГОСЯ ТИПА
' =====================================

Function RemoveType(s As String, _
                    code As String) As String

    Dim prefix As String

    Select Case code

        Case "C"
            prefix = "Конденсатор"

        Case "R"
            prefix = "Резистор"

        Case "L"
            prefix = "Катушка индуктивности"

        Case "VD"
            prefix = "Диод"

        Case "VT"
            prefix = "Транзистор"

        Case "DA", "DD", "U"
            prefix = "Микросхема"

        Case Else
            prefix = ""

    End Select

    If prefix <> "" Then

        If StrComp(Left(s, Len(prefix) + 1), _
           prefix & " ", vbTextCompare) = 0 Then

            s = Mid(s, Len(prefix) + 2)

        End If

    End If

    RemoveType = s

End Function


' =====================================
' НАЗВАНИЯ ГРУПП
' =====================================

Function GroupName(code As String) As String

    Select Case code

        Case "C"
            GroupName = "КОНДЕНСАТОРЫ"

        Case "R"
            GroupName = "РЕЗИСТОРЫ"

        Case "L"
            GroupName = "КАТУШКИ ИНДУКТИВНОСТИ"

        Case "VD"
            GroupName = "ДИОДЫ"

        Case "VT"
            GroupName = "ТРАНЗИСТОРЫ"

        Case "DA", "DD", "U"
            GroupName = "МИКРОСХЕМЫ"

        Case "X", "XP", "XS"
            GroupName = "СОЕДИНИТЕЛИ"

        Case "ZQ"
            GroupName = "РЕЗОНАТОРЫ"

        Case Else
            GroupName = "ЭЛЕМЕНТЫ " & code

    End Select

End Function
