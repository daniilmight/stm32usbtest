
Option Explicit

Private Type PEItem
    Code As String
    Num As Long
    Model As String
    Maker As String
End Type

Private Const LINE_H As Single = 22.5
Private Const NAME_W As Single = 315


' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Sub MakeWordPE()

    Dim xl As Object, book As Object, sh As Object
    Dim doc As Document, t As Table
    Dim a As Variant
    Dim el() As PEItem, tmp As PEItem

    Dim file As String
    Dim last As Long, total As Long
    Dim i As Long, j As Long
    Dim p As Long, num As Long
    Dim n1 As Long, n2 As Long
    Dim page As Long, used As Long
    Dim limit As Long, rowsUsed As Long
    Dim lines As Long, need As Long

    Dim ref As String, code As String
    Dim model As String, maker As String
    Dim grp As String, oldGrp As String
    Dim firstRef As String, lastRef As String
    Dim parts As Variant, part As Variant

    On Error GoTo ErrorHandler

    ' Выбор файла Excel

    With Application.FileDialog(3)
        .Title = "Выберите файл Excel"
        .Filters.Clear
        .Filters.Add "Excel", "*.xlsx;*.xlsm;*.xls"

        If .Show <> -1 Then Exit Sub
        file = .SelectedItems(1)
    End With

    ' Чтение данных Excel

    Set xl = CreateObject("Excel.Application")
    Set book = xl.Workbooks.Open(file, 0, True)
    Set sh = book.Worksheets(1)

    last = sh.Cells(sh.Rows.Count, 1).End(-4162).Row

    If last < 2 Then
        MsgBox "Нет данных."
        GoTo Finish
    End If

    a = sh.Range("A2:C" & last).Value2

    book.Close False
    Set book = Nothing
    xl.Quit
    Set xl = Nothing

    ' =====================================
    ' РАЗБОР ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
    ' =====================================

    total = 0

    For i = 1 To UBound(a, 1)

        SplitName CStr(a(i, 2)), model, maker

        ref = Trim(CStr(a(i, 1)))
        ref = Replace(ref, "–", "-")
        ref = Replace(ref, "—", "-")
        ref = Replace(ref, ";", ",")

        parts = Split(ref, ",")

        For Each part In parts

            ref = Trim(CStr(part))

            If ref <> "" Then

                p = InStr(ref, "-")

                If p > 0 Then
                    firstRef = Trim(Left(ref, p - 1))
                    lastRef = Trim(Mid(ref, p + 1))
                Else
                    firstRef = ref
                    lastRef = ref
                End If

                code = RefCode(firstRef)
                n1 = RefNum(firstRef)
                n2 = RefNum(lastRef)

                If code = "" Or n1 < 1 Or n2 < n1 Then
                    MsgBox "Ошибка обозначения: " & ref
                    GoTo Finish
                End If

                For num = n1 To n2

                    total = total + 1
                    ReDim Preserve el(1 To total)

                    el(total).Code = code
                    el(total).Num = num
                    el(total).Model = model
                    el(total).Maker = maker

                Next num

            End If

        Next part

    Next i

    If total = 0 Then GoTo Finish

    ' =====================================
    ' СОРТИРОВКА
    ' =====================================

    For i = 1 To total - 1
        For j = i + 1 To total

            If el(j).Code < el(i).Code Or _
               (el(j).Code = el(i).Code And _
                el(j).Num < el(i).Num) Then

                tmp = el(i)
                el(i) = el(j)
                el(j) = tmp

            End If

        Next j
    Next i

    ' Проверка повторяющихся обозначений

    For i = 2 To total

        If el(i).Code = el(i - 1).Code And _
           el(i).Num = el(i - 1).Num Then

            MsgBox "Повтор обозначения: " & _
                   el(i).Code & el(i).Num
            GoTo Finish

        End If

    Next i

    ' =====================================
    ' СОЗДАНИЕ ДОКУМЕНТА
    ' =====================================

    Set doc = Documents.Add

    With doc.PageSetup
        .PaperSize = wdPaperA4
        .Orientation = wdOrientPortrait
        .TopMargin = CentimetersToPoints(1.2)
        .BottomMargin = CentimetersToPoints(1.2)
        .LeftMargin = CentimetersToPoints(1)
        .RightMargin = CentimetersToPoints(1)
    End With

    page = 1
    limit = 23
    used = 0
    rowsUsed = 0
    oldGrp = ""

    Set t = CreatePETable(doc)

    ' =====================================
    ' ФОРМИРОВАНИЕ ПЕРЕЧНЯ
    ' =====================================

    i = 1

    Do While i <= total

        j = i

        ' Объединение соседних одинаковых
        ' элементов

        Do While j < total

            If el(j + 1).Code <> el(i).Code Then
                Exit Do
            End If

            If el(j + 1).Num <> el(j).Num + 1 Then
                Exit Do
            End If

            If el(j + 1).Model <> el(i).Model Or _
               el(j + 1).Maker <> el(i).Maker Then
                Exit Do
            End If

            j = j + 1

        Loop

        code = el(i).Code
        model = RemoveType(el(i).Model, code)
        maker = el(i).Maker

        If i = j Then
            ref = code & el(i).Num
        Else
            ref = code & el(i).Num & "-" & _
                  code & el(j).Num
        End If

        grp = GroupName(code)

        ' Измерение числа строк текста

        lines = MeasureName(t, model, maker)

        If lines > 27 Then
            MsgBox "Слишком длинная запись: " & ref
            GoTo Finish
        End If

        need = lines

        If grp <> oldGrp Then
            need = need + 3
        End If

        ' Проверка заполнения страницы

        If used + need > limit Then

            NextPEPage doc, t, page, used, _
                       limit, rowsUsed

        End If

        ' Заголовок группы

        If grp <> oldGrp Then

            AddBlank t, rowsUsed
            used = used + 1

            AddGroup t, rowsUsed, grp
            used = used + 1

            AddBlank t, rowsUsed
            used = used + 1

            oldGrp = grp

        End If

        ' Добавление элемента

        AddItem t, rowsUsed, ref, model, _
                maker, j - i + 1, lines

        used = used + lines

        i = j + 1

    Loop

    ' Пустые строки в конце

    Do While used < limit
        AddBlank t, rowsUsed
        used = used + 1
    Loop

    MsgBox "Перечень готов. Страниц: " & page

Finish:

    On Error Resume Next

    If Not book Is Nothing Then book.Close False
    If Not xl Is Nothing Then xl.Quit

    Exit Sub

ErrorHandler:

    MsgBox "Ошибка: " & Err.Description
    Resume Finish

End Sub


' =====================================
' СОЗДАНИЕ ТАБЛИЦЫ WORD
' =====================================

Function CreatePETable(doc As Document) As Table

    Dim r As Range
    Dim t As Table

    Set r = doc.Content
    r.Collapse wdCollapseEnd

    Set t = doc.Tables.Add(r, 1, 4)

    t.AllowAutoFit = False

    t.Columns(1).Width = 82
    t.Columns(2).Width = NAME_W
    t.Columns(3).Width = 38
    t.Columns(4).Width = 95

    With t
        .Borders.Enable = True
        .TopPadding = 0
        .BottomPadding = 0
        .LeftPadding = 3
        .RightPadding = 3
    End With

    With t.Range

        .Font.Name = "Arial"
        .Font.Size = 12

        .ParagraphFormat.SpaceBefore = 0
        .ParagraphFormat.SpaceAfter = 0

        .ParagraphFormat.LineSpacingRule = _
            wdLineSpaceExactly

        .ParagraphFormat.LineSpacing = 18

    End With

    t.Rows.HeightRule = wdRowHeightAtLeast
    t.Rows.Height = LINE_H
    t.Rows.AllowBreakAcrossPages = False

    Set CreatePETable = t

End Function


' =====================================
' ДОБАВЛЕНИЕ СТРОКИ
' =====================================

Sub AddPERow(t As Table, ByRef rowsUsed As Long)

    If rowsUsed > 0 Then t.Rows.Add

    rowsUsed = rowsUsed + 1

    With t.Rows(rowsUsed)
        .HeightRule = wdRowHeightAtLeast
        .Height = LINE_H
        .AllowBreakAcrossPages = False
    End With

End Sub


' =====================================
' ПУСТАЯ СТРОКА
' =====================================

Sub AddBlank(t As Table, ByRef rowsUsed As Long)

    AddPERow t, rowsUsed

End Sub


' =====================================
' ЗАГОЛОВОК ГРУППЫ
' =====================================

Sub AddGroup(t As Table, _
             ByRef rowsUsed As Long, _
             title As String)

    AddPERow t, rowsUsed

    With t.Cell(rowsUsed, 2).Range

        .Text = title
        .Font.Name = "Arial"
        .Font.Size = 12

    End With

End Sub


' =====================================
' НОВАЯ СТРАНИЦА
' =====================================

Sub NextPEPage(doc As Document, _
               ByRef t As Table, _
               ByRef page As Long, _
               ByRef used As Long, _
               ByRef limit As Long, _
               ByRef rowsUsed As Long)

    Dim r As Range

    Do While used < limit
        AddBlank t, rowsUsed
        used = used + 1
    Loop

    Set r = doc.Range(doc.Content.End - 1, _
                      doc.Content.End - 1)

    r.InsertBreak wdPageBreak

    Set t = CreatePETable(doc)

    page = page + 1
    limit = 30
    used = 0
    rowsUsed = 0

End Sub


' =====================================
' РАЗМЕЩЕНИЕ МОДЕЛИ И ФИРМЫ/ТУ
' =====================================

Function PutName(c As Cell, _
                 model As String, _
                 maker As String) As Long

    Dim r As Range
    Dim lines As Long

    If maker = "" Then

        c.Range.Text = model

    Else

        ' Пробуем разместить в одну строку

        c.Range.Text = model & vbTab & maker

        Set r = c.Range
        r.MoveEnd wdCharacter, -1

        With r.ParagraphFormat.TabStops

            .ClearAll

            .Add Position:=NAME_W - 8, _
                 Alignment:=wdAlignTabRight

        End With

        c.Range.Document.Repaginate

        lines = r.ComputeStatistics(wdStatisticLines)

        ' Если не поместилось,
        ' переносим фирму/ТУ вниз

        If lines > 1 Then

            c.Range.Text = model & vbCr & maker

            c.Range.Paragraphs(1).Alignment = _
                wdAlignParagraphLeft

            c.Range.Paragraphs(2).Alignment = _
                wdAlignParagraphRight

        End If

    End If

    ' Итоговое количество строк

    Set r = c.Range
    r.MoveEnd wdCharacter, -1

    c.Range.Document.Repaginate

    lines = r.ComputeStatistics(wdStatisticLines)

    If lines < 1 Then lines = 1

    PutName = lines

End Function


' =====================================
' ИЗМЕРЕНИЕ ЧИСЛА СТРОК
' =====================================

Function MeasureName(t As Table, _
                     model As String, _
                     maker As String) As Long

    Dim rw As Row

    ' Добавляем временную строку

    Set rw = t.Rows.Add

    MeasureName = PutName(rw.Cells(2), _
                          model, maker)

    ' Удаляем её после измерения

    rw.Delete

End Function


' =====================================
' ДОБАВЛЕНИЕ ЭЛЕМЕНТА
' =====================================

Sub AddItem(t As Table, _
            ByRef rowsUsed As Long, _
            ref As String, _
            model As String, _
            maker As String, _
            qty As Long, _
            lines As Long)

    Dim realLines As Long

    AddPERow t, rowsUsed

    t.Cell(rowsUsed, 1).Range.Text = ref

    realLines = PutName(t.Cell(rowsUsed, 2), _
                        model, maker)

    t.Rows(rowsUsed).Height = _
        realLines * LINE_H

    ' Количество снизу

    With t.Cell(rowsUsed, 3)

        .Range.Text = CStr(qty)

        .Range.ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

        .VerticalAlignment = _
            wdCellAlignVerticalBottom

    End With

    t.Cell(rowsUsed, 1).VerticalAlignment = _
        wdCellAlignVerticalTop

End Sub


' =====================================
' БУКВЕННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefCode(s As String) As String

    Dim i As Long

    s = UCase(Trim(s))

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then Exit For

    Next i

    RefCode = Left(s, i - 1)

End Function


' =====================================
' НОМЕР ЭЛЕМЕНТА
' =====================================

Function RefNum(s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function


' =====================================
' РАЗДЕЛЕНИЕ МОДЕЛИ И ФИРМЫ/ТУ
' =====================================

Sub SplitName(s As String, _
              ByRef model As String, _
              ByRef maker As String)

    Dim p As Long, word As String

    s = Trim(s)
    model = s
    maker = ""

    p = InStr(1, s, " ф. ", vbTextCompare)

    If p > 0 Then

        model = Trim(Left(s, p - 1))
        maker = Trim(Mid(s, p + 1))

    Else

        p = InStrRev(s, " ")

        If p > 0 Then

            word = Trim(Mid(s, p + 1))

            If Len(word) > 2 And _
               Right(UCase(word), 2) = "ТУ" And _
               InStr(word, ".") > 0 Then

                model = Trim(Left(s, p - 1))
                maker = word

            End If

        End If

    End If

End Sub


' =====================================
' УДАЛЕНИЕ НАЗВАНИЯ ТИПА ЭЛЕМЕНТА
' =====================================

Function RemoveType(s As String, _
                    code As String) As String

    Dim prefix As String

    Select Case code

        Case "C"
            prefix = "Конденсатор"

        Case "R"
            prefix = "Резистор"

        Case "L"
            prefix = "Катушка индуктивности"

        Case "VD"
            prefix = "Диод"

        Case "VT"
            prefix = "Транзистор"

        Case "DA", "DD", "U"
            prefix = "Микросхема"

        Case Else
            prefix = ""

    End Select

    If prefix <> "" Then

        If StrComp(Left(s, Len(prefix) + 1), _
           prefix & " ", vbTextCompare) = 0 Then

            s = Mid(s, Len(prefix) + 2)

        End If

    End If

    RemoveType = s

End Function


' =====================================
' НАЗВАНИЯ ГРУПП
' =====================================

Function GroupName(code As String) As String

    Select Case code

        Case "C"
            GroupName = "КОНДЕНСАТОРЫ"

        Case "R"
            GroupName = "РЕЗИСТОРЫ"

        Case "L"
            GroupName = "КАТУШКИ ИНДУКТИВНОСТИ"

        Case "VD"
            GroupName = "ДИОДЫ"

        Case "VT"
            GroupName = "ТРАНЗИСТОРЫ"

        Case "DA", "DD", "U"
            GroupName = "МИКРОСХЕМЫ"

        Case "X", "XP", "XS"
            GroupName = "СОЕДИНИТЕЛИ"

        Case "ZQ"
            GroupName = "РЕЗОНАТОРЫ"

        Case Else
            GroupName = "ЭЛЕМЕНТЫ " & code

    End Select

End Function
