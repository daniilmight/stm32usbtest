
Option Explicit

' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Sub MakePE()

    Dim src As Worksheet, dst As Worksheet
    Dim wb As Workbook
    Dim a As Variant, temp As Variant
    Dim i As Long, j As Long, k As Long
    Dim last As Long, page As Long
    Dim r As Long, maxRow As Long, n As Long

    Dim ref As String, code As String
    Dim oldCode As String
    Dim model As String, maker As String
    Dim fullName As String, prefix As String
    Dim lastWord As String
    Dim p As Long

    Dim refs As Collection
    Dim names As Collection
    Dim makers As Collection

    Set src = ActiveSheet

    last = src.Cells(src.Rows.Count, 1).End(xlUp).Row
    If last < 2 Then Exit Sub

    a = src.Range("A2:C" & last).Value

    ' Сортировка по обозначению и номеру

    For i = 1 To UBound(a, 1) - 1
        For j = i + 1 To UBound(a, 1)

            If RefCode(CStr(a(j, 1))) < _
               RefCode(CStr(a(i, 1))) Or _
               (RefCode(CStr(a(j, 1))) = _
               RefCode(CStr(a(i, 1))) And _
               RefNum(CStr(a(j, 1))) < _
               RefNum(CStr(a(i, 1)))) Then

                For k = 1 To 3
                    temp = a(i, k)
                    a(i, k) = a(j, k)
                    a(j, k) = temp
                Next k

            End If

        Next j
    Next i

    ' Создание новой книги

    Set wb = Workbooks.Add(xlWBATWorksheet)
    Set dst = wb.Worksheets(1)

    page = 1
    r = 2
    maxRow = 24
    oldCode = ""

    PreparePage dst, page

    ' Обработка элементов

    For i = 1 To UBound(a, 1)

        ref = Trim(CStr(a(i, 1)))

        If ref <> "" Then

            code = RefCode(ref)
            fullName = Trim(CStr(a(i, 2)))

            ' =================================
            ' ИЗМЕНЕНИЕ 1
            ' Определение изготовителя или ТУ
            ' =================================

            model = fullName
            maker = ""

            ' Поиск обозначения "ф. "

            p = InStr(1, fullName, _
                      " ф. ", vbTextCompare)

            If p > 0 Then

                model = Trim(Left(fullName, p - 1))
                maker = Trim(Mid(fullName, p + 1))

            Else

                ' Поиск ТУ в конце строки

                p = InStrRev(fullName, " ")

                If p > 0 Then

                    lastWord = Trim(Mid(fullName, p + 1))

                    If Len(lastWord) > 2 And _
                       Right(UCase(lastWord), 2) = "ТУ" _
                       And InStr(lastWord, ".") > 0 Then

                        model = Trim(Left(fullName, p - 1))
                        maker = lastWord

                    End If

                End If

            End If

            ' =================================
            ' Удаление повторяющегося типа
            ' =================================

            prefix = ""

            Select Case code
                Case "C"
                    prefix = "Конденсатор"
                Case "R"
                    prefix = "Резистор"
                Case "L"
                    prefix = "Катушка индуктивности"
                Case "VD"
                    prefix = "Диод"
                Case "VT"
                    prefix = "Транзистор"
                Case "DA", "DD"
                    prefix = "Микросхема"
            End Select

            If prefix <> "" Then

                If StrComp(Left(model, Len(prefix) + 1), _
                   prefix & " ", vbTextCompare) = 0 Then

                    model = Mid(model, Len(prefix) + 2)

                End If

            End If

            ' =================================
            ' ИЗМЕНЕНИЕ 2
            ' Длина зависит от наличия фирмы
            ' =================================

            Set refs = SplitLines(ref, 10)

            If maker = "" Then
                Set names = SplitLines(model, 48)
            Else
                Set names = SplitLines(model, 32)
            End If

            Set makers = SplitLines(maker, 14)

            n = Application.Max(refs.Count, _
                                names.Count, makers.Count)

            ' =================================
            ' Начало новой группы
            ' =================================

            If code <> oldCode Then

                ' ИЗМЕНЕНИЕ 3
                ' Пустая строка перед заголовком,
                ' заголовок и пустая строка после

                k = 3 + n

                If r + k - 1 > maxRow Then

                    NewPage wb, dst, page, r, maxRow

                End If

                r = r + 1

                dst.Cells(r, 2).Value = GroupName(code)

                dst.Range(dst.Cells(r, 2), _
                          dst.Cells(r, 3)).Merge

                r = r + 2
                oldCode = code

            ElseIf r + n - 1 > maxRow Then

                NewPage wb, dst, page, r, maxRow

            End If

            ' =================================
            ' Заполнение строк элемента
            ' =================================

            For k = 1 To n

                If k <= refs.Count Then
                    dst.Cells(r, 1).Value = refs(k)
                End If

                If k <= names.Count Then
                    dst.Cells(r, 2).Value = names(k)
                End If

                If k <= makers.Count Then
                    dst.Cells(r, 3).Value = makers(k)
                End If

                ' ИЗМЕНЕНИЕ 4
                ' Объединение при отсутствии
                ' изготовителя или ТУ

                If maker = "" Then

                    dst.Range(dst.Cells(r, 2), _
                              dst.Cells(r, 3)).Merge

                End If

                If k = 1 Then
                    dst.Cells(r, 4).Value = a(i, 3)
                End If

                r = r + 1

            Next k

        End If

    Next i

    MsgBox "Перечень готов. Страниц: " & page

End Sub


' =====================================
' ОФОРМЛЕНИЕ СТРАНИЦЫ
' =====================================

Sub PreparePage(ws As Worksheet, page As Long)

    Dim lastRow As Long

    If page = 1 Then
        lastRow = 24
    Else
        lastRow = 31
    End If

    ws.Name = "ПЭ_" & page

    ' Ширина столбцов

    ws.Columns("A").ColumnWidth = 11
    ws.Columns("B").ColumnWidth = 40
    ws.Columns("C").ColumnWidth = 19
    ws.Columns("D").ColumnWidth = 5.5
    ws.Columns("E").ColumnWidth = 24

    ' Общее оформление

    With ws.Range("A1:E" & lastRow)

        .Font.Name = "Arial"
        .Font.Size = 12
        .RowHeight = 22.5
        .VerticalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous

    End With

    ' Заголовки столбцов

    ws.Range("B1:C1").Merge

    ws.Range("A1").Value = "Поз. обозначение"
    ws.Range("B1").Value = "Наименование"
    ws.Range("D1").Value = "Кол."
    ws.Range("E1").Value = "Примечание"

    ws.Rows(1).RowHeight = 36

    ws.Range("A1:E1").HorizontalAlignment = xlCenter
    ws.Range("A1:E1").WrapText = True

    ws.Columns("A").HorizontalAlignment = xlCenter
    ws.Columns("D").HorizontalAlignment = xlCenter

    ' Настройки печати

    With ws.PageSetup

        .PaperSize = xlPaperA4
        .Orientation = xlPortrait

        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = 1

        .PrintArea = "A1:E" & lastRow

    End With

End Sub


' =====================================
' СОЗДАНИЕ СЛЕДУЮЩЕЙ СТРАНИЦЫ
' =====================================

Sub NewPage(wb As Workbook, ws As Worksheet, _
            ByRef page As Long, ByRef r As Long, _
            ByRef maxRow As Long)

    page = page + 1

    Set ws = wb.Worksheets.Add( _
        After:=wb.Worksheets(wb.Worksheets.Count))

    PreparePage ws, page

    r = 2
    maxRow = 31

End Sub


' =====================================
' ОПРЕДЕЛЕНИЕ БУКВЕННОГО ОБОЗНАЧЕНИЯ
' =====================================

Function RefCode(s As String) As String

    Dim i As Long

    s = Trim(UCase(s))

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then Exit For

    Next i

    RefCode = Left(s, i - 1)

End Function


' =====================================
' ОПРЕДЕЛЕНИЕ ПЕРВОГО НОМЕРА
' =====================================

Function RefNum(s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function


' =====================================
' НАЗВАНИЯ ГРУПП ЭЛЕМЕНТОВ
' =====================================

Function GroupName(code As String) As String

    Select Case code

        Case "C"
            GroupName = "КОНДЕНСАТОРЫ"

        Case "R"
            GroupName = "РЕЗИСТОРЫ"

        Case "L"
            GroupName = "КАТУШКИ ИНДУКТИВНОСТИ"

        Case "VD"
            GroupName = "ДИОДЫ"

        Case "VT"
            GroupName = "ТРАНЗИСТОРЫ"

        Case "DA", "DD", "U"
            GroupName = "МИКРОСХЕМЫ"

        Case "X", "XP", "XS"
            GroupName = "СОЕДИНИТЕЛИ"

        Case "ZQ"
            GroupName = "РЕЗОНАТОРЫ"

        Case Else
            GroupName = "ЭЛЕМЕНТЫ " & code

    End Select

End Function


' =====================================
' ПЕРЕНОС ДЛИННОГО ТЕКСТА
' =====================================

Function SplitLines(s As String, _
                    maxLen As Long) As Collection

    Dim result As New Collection
    Dim words As Variant, word As Variant
    Dim line As String, piece As String

    words = Split(Trim(s), " ")

    For Each word In words

        piece = CStr(word)

        ' Разделение слишком длинного слова

        Do While Len(piece) > maxLen

            If line <> "" Then
                result.Add line
                line = ""
            End If

            result.Add Left(piece, maxLen)
            piece = Mid(piece, maxLen + 1)

        Loop

        ' Проверка длины строки

        If Len(line) + Len(piece) + 1 > maxLen Then

            If line <> "" Then result.Add line
            line = ""

        End If

        If line <> "" Then line = line & " "
        line = line & piece

    Next word

    If line <> "" Then result.Add line

    Set SplitLines = result

End Function
