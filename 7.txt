
Option Explicit

' Данные отдельного элемента

Private Type PEItem
    Code As String
    Num As Long
    Model As String
    Maker As String
    TokenId As Long
End Type

Private Const FIRST_ROWS As Long = 23
Private Const NEXT_ROWS As Long = 30


' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Sub MakeWordPE()

    Dim xl As Object, book As Object, sh As Object
    Dim doc As Document, testDoc As Document
    Dim testTable As Table

    Dim a As Variant
    Dim el() As PEItem, tmp As PEItem
    Dim parts As Variant, part As Variant
    Dim pair As Variant, piece As Variant

    Dim pages As Collection
    Dim pageRows As Collection
    Dim chunks As Collection
    Dim refLines As Collection
    Dim textLines As Collection

    Dim templateFile As String
    Dim excelFile As String
    Dim outputFile As String, baseName As String

    Dim ref As String, code As String
    Dim endCode As String
    Dim firstRef As String, lastRef As String
    Dim model As String, maker As String
    Dim grp As String, oldGrp As String
    Dim refText As String, qtyText As String

    Dim last As Long, total As Long
    Dim i As Long, j As Long, k As Long
    Dim num As Long, n1 As Long, n2 As Long
    Dim pos As Long, n As Long
    Dim need As Long, limit As Long
    Dim pg As Long, token As Long
    Dim newGroup As Boolean

    On Error GoTo ErrorHandler

    ' =====================================
    ' ВЫБОР ШАБЛОНА WORD
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите шаблон Word"
        .Filters.Clear
        .Filters.Add "Word", "*.doc;*.docx;*.docm"

        If .Show <> -1 Then Exit Sub

        templateFile = .SelectedItems(1)

    End With

    ' =====================================
    ' ВЫБОР EXCEL
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите перечень Excel"
        .Filters.Clear
        .Filters.Add "Excel", "*.xlsx;*.xlsm;*.xls"

        If .Show <> -1 Then Exit Sub

        excelFile = .SelectedItems(1)

    End With

    ' Чтение Excel

    Set xl = CreateObject("Excel.Application")

    Set book = xl.Workbooks.Open( _
        excelFile, 0, True)

    Set sh = book.Worksheets(1)

    last = sh.Cells(sh.Rows.Count, 1).End(-4162).Row

    If last < 2 Then
        MsgBox "В Excel нет данных."
        GoTo Finish
    End If

    a = sh.Range("A2:C" & last).Value2

    book.Close False
    Set book = Nothing

    xl.Quit
    Set xl = Nothing


    ' =====================================
    ' РАЗБОР ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
    ' =====================================

    total = 0
    token = 0

    For i = 1 To UBound(a, 1)

        SplitName CStr(a(i, 2)), model, maker

        ref = Trim(CStr(a(i, 1)))

        ref = Replace(ref, "–", "-")
        ref = Replace(ref, "—", "-")
        ref = Replace(ref, ",", " ")
        ref = Replace(ref, ";", " ")
        ref = Replace(ref, vbTab, " ")
        ref = Replace(ref, Chr(160), " ")

        parts = Split(ref, " ")

        For Each part In parts

            ref = Trim(CStr(part))

            If ref <> "" Then

                ' Каждый исходный фрагмент
                ' получает собственный номер.
                ' Это позволяет отличить
                ' C5 C6 от диапазона C5-C6.

                token = token + 1

                pos = InStr(ref, "-")

                If pos > 0 Then

                    firstRef = Left(ref, pos - 1)
                    lastRef = Mid(ref, pos + 1)

                Else

                    firstRef = ref
                    lastRef = ref

                End If

                code = RefCode(firstRef)
                endCode = RefCode(lastRef)

                n1 = RefNum(firstRef)
                n2 = RefNum(lastRef)

                If code = "" Or n1 < 1 Or _
                   n2 < n1 Or _
                   (endCode <> "" And endCode <> code) Then

                    Err.Raise vbObjectError + 100, , _
                        "Ошибка обозначения: " & ref

                End If

                For num = n1 To n2

                    total = total + 1

                    ReDim Preserve el(1 To total)

                    el(total).Code = code
                    el(total).Num = num
                    el(total).Model = model
                    el(total).Maker = maker
                    el(total).TokenId = token

                Next num

            End If

        Next part

    Next i

    If total = 0 Then GoTo Finish


    ' =====================================
    ' СОРТИРОВКА ОБОЗНАЧЕНИЙ
    ' =====================================

    For i = 1 To total - 1

        For j = i + 1 To total

            If el(j).Code < el(i).Code Or _
               (el(j).Code = el(i).Code And _
                el(j).Num < el(i).Num) Then

                tmp = el(i)
                el(i) = el(j)
                el(j) = tmp

            End If

        Next j

    Next i

    ' Проверка повторов

    For i = 2 To total

        If el(i).Code = el(i - 1).Code And _
           el(i).Num = el(i - 1).Num Then

            Err.Raise vbObjectError + 101, , _
                "Повтор обозначения: " & _
                el(i).Code & el(i).Num

        End If

    Next i


    ' =====================================
    ' ВРЕМЕННАЯ ТАБЛИЦА ДЛЯ ИЗМЕРЕНИЙ
    ' =====================================

    Set testDoc = Documents.Add(Visible:=False)

    SetupPEPage testDoc.Sections(1)

    Set testTable = testDoc.Tables.Add( _
        testDoc.Range(0, 0), 1, 4)

    FormatPETable testTable, 1

    ' При измерении допускаем
    ' автоматическое увеличение высоты

    testTable.Rows.HeightRule = wdRowHeightAtLeast


    ' =====================================
    ' ПОДГОТОВКА СТРАНИЦ
    ' =====================================

    Set pages = New Collection
    Set pageRows = New Collection

    pages.Add pageRows

    limit = FIRST_ROWS
    oldGrp = ""

    i = 1

    Do While i <= total

        j = i

        ' =====================================
        ' ОБЪЕДИНЕНИЕ ОДИНАКОВЫХ ЭЛЕМЕНТОВ
        ' =====================================

        Do While j < total

            If el(j + 1).Code <> el(i).Code Then
                Exit Do
            End If

            If el(j + 1).Num <> el(j).Num + 1 Then
                Exit Do
            End If

            If el(j + 1).Model <> el(i).Model Or _
               el(j + 1).Maker <> el(i).Maker Then
                Exit Do
            End If

            j = j + 1

        Loop

        code = el(i).Code
        model = RemoveType(el(i).Model, code)
        maker = el(i).Maker

        ' =====================================
        ' ИЗМЕНЕНИЕ 1
        ' ЗАПЯТАЯ ВМЕСТО АВТОМАТИЧЕСКОГО ТИРЕ
        ' =====================================

        ref = BuildReference(el, i, j)

        ' =====================================
        ' ИЗМЕНЕНИЕ 2
        ' ГРУППИРУЕМ ТОЛЬКО C, R, DA, DD, U
        ' =====================================

        grp = GroupName(code)

        newGroup = (grp <> "" And grp <> oldGrp)


        ' =====================================
        ' ФОРМИРОВАНИЕ НАИМЕНОВАНИЯ
        ' =====================================

        Set chunks = New Collection

        ' Модель и фирма в одной строке,
        ' только если есть запас три пробела

        If maker <> "" And _
           TextFits(testTable, _
                    model & Space(3) & maker, 2) Then

            chunks.Add Array(model, maker)

        Else

            ' Модель слева

            Set textLines = WrapText( _
                testTable, model, 2)

            For Each piece In textLines

                chunks.Add Array(CStr(piece), "")

            Next piece

            ' Фирма или ТУ справа

            Set textLines = WrapText( _
                testTable, maker, 2)

            For Each piece In textLines

                chunks.Add Array("", CStr(piece))

            Next piece

        End If

        If chunks.Count = 0 Then
            chunks.Add Array("", "")
        End If


        ' =====================================
        ' ИЗМЕНЕНИЕ 3
        ' ПЕРЕНОС ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
        ' =====================================

        Set refLines = WrapReference(testTable, ref)

        n = chunks.Count

        If refLines.Count > n Then
            n = refLines.Count
        End If


        ' =====================================
        ' РАСПРЕДЕЛЕНИЕ ПО СТРАНИЦАМ
        ' =====================================

        need = n

        If newGroup Then
            need = need + 3
        End If

        If need > NEXT_ROWS Then

            Err.Raise vbObjectError + 102, , _
                "Слишком длинная запись: " & ref

        End If

        If pageRows.Count + need > limit Then

            If pageRows.Count = 0 Then

                Err.Raise vbObjectError + 103, , _
                    "Запись не помещается " & _
                    "на пустой лист: " & ref

            End If

            FillPage pageRows, limit

            Set pageRows = New Collection
            pages.Add pageRows

            limit = NEXT_ROWS

        End If


        ' =====================================
        ' ЗАГОЛОВОК ТОЛЬКО НУЖНОЙ ГРУППЫ
        ' =====================================

        If newGroup Then

            ' Пустая строка перед заголовком

            pageRows.Add Array("", "", "", "")

            ' Заголовок

            pageRows.Add Array("", grp, "", "")

            ' Пустая строка после заголовка

            pageRows.Add Array("", "", "", "")

        End If

        oldGrp = grp


        ' =====================================
        ' ЗАПИСЬ ЭЛЕМЕНТА
        ' =====================================

        For k = 1 To n

            If k <= chunks.Count Then

                pair = chunks(k)

            Else

                pair = Array("", "")

            End If

            refText = ""
            qtyText = ""

            ' Обозначение может занимать
            ' несколько физических строк

            If k <= refLines.Count Then
                refText = CStr(refLines(k))
            End If

            ' Количество в последней строке

            If k = n Then
                qtyText = CStr(j - i + 1)
            End If

            pageRows.Add Array( _
                refText, _
                CStr(pair(0)), _
                CStr(pair(1)), _
                qtyText)

        Next k

        i = j + 1

    Loop

    ' Дополнение последней страницы

    FillPage pageRows, limit

    testDoc.Close wdDoNotSaveChanges
    Set testDoc = Nothing


    ' =====================================
    ' ОТКРЫТИЕ ШАБЛОНА WORD
    ' =====================================

    Set doc = Documents.Open( _
        FileName:=templateFile, _
        ReadOnly:=True, _
        AddToRecentFiles:=False)

    If doc.Sections.Count <> 3 Then

        Err.Raise vbObjectError + 104, , _
            "В шаблоне должно быть 3 раздела." & _
            vbCrLf & "Обнаружено: " & _
            doc.Sections.Count

    End If

    ' Формирование имени выходного файла

    baseName = Left(templateFile, _
        InStrRev(templateFile, ".") - 1)

    outputFile = baseName & "_PE.docx"

    k = 1

    Do While Dir(outputFile) <> ""

        outputFile = baseName & "_PE_" & k & ".docx"
        k = k + 1

    Loop

    ' Сохраняем копию

    doc.SaveAs2 FileName:=outputFile, _
                FileFormat:=wdFormatXMLDocument


    ' =====================================
    ' ПОДГОТОВКА РАЗДЕЛОВ
    ' =====================================

    ' Удаляем прежнее содержимое
    ' первого и второго разделов

    ClearSectionBody doc, 2
    ClearSectionBody doc, 1

    ' Раздел 3 не очищаем

    doc.Sections(2).PageSetup.SectionStart = _
        wdSectionNewPage

    doc.Sections(3).PageSetup.SectionStart = _
        wdSectionNewPage

    ' Поля первых двух разделов - нулевые

    SetupPEPage doc.Sections(1)
    SetupPEPage doc.Sections(2)


    ' =====================================
    ' СОЗДАНИЕ ТАБЛИЦ ПО СТРАНИЦАМ
    ' =====================================

    For pg = 1 To pages.Count

        Set pageRows = pages(pg)

        WritePEPage doc, pageRows, pg

    Next pg

    doc.Repaginate
    doc.Save

    MsgBox "Перечень готов!" & vbCrLf & _
           "Листов перечня: " & pages.Count & _
           vbCrLf & "Файл: " & outputFile

Finish:

    On Error Resume Next

    If Not book Is Nothing Then book.Close False
    If Not xl Is Nothing Then xl.Quit

    If Not testDoc Is Nothing Then
        testDoc.Close wdDoNotSaveChanges
    End If

    Exit Sub

ErrorHandler:

    MsgBox "Ошибка: " & Err.Description

    On Error Resume Next

    If Not doc Is Nothing Then
        doc.Close wdDoNotSaveChanges
    End If

    Resume Finish

End Sub


' =====================================
' ИЗМЕНЕНИЕ 4
' СОСТАВЛЕНИЕ ПОЗИЦИОННОГО ОБОЗНАЧЕНИЯ
'
' C5 C6 -> C5, C6
' C5-C6 -> C5-C6
' =====================================

Function BuildReference(el() As PEItem, _
                        first As Long, _
                        last As Long) As String

    Dim k As Long, j As Long
    Dim result As String
    Dim part As String

    result = ""
    k = first

    Do While k <= last

        j = k

        ' Диапазон допускается только
        ' внутри исходного фрагмента

        Do While j < last

            If el(j + 1).TokenId <> _
               el(k).TokenId Then Exit Do

            If el(j + 1).Num <> _
               el(j).Num + 1 Then Exit Do

            j = j + 1

        Loop

        If j = k Then

            part = el(k).Code & el(k).Num

        Else

            part = el(k).Code & el(k).Num & _
                   "-" & el(j).Code & el(j).Num

        End If

        If result <> "" Then
            result = result & ", "
        End If

        result = result & part

        k = j + 1

    Loop

    BuildReference = result

End Function


' =====================================
' СОЗДАНИЕ ТАБЛИЦЫ НА СТРАНИЦЕ
' =====================================

Sub WritePEPage(doc As Document, _
                rows As Collection, _
                pageNo As Long)

    Dim rg As Range
    Dim t As Table
    Dim data As Variant

    Dim r As Long
    Dim pos As Long
    Dim sectionNum As Long

    If pageNo = 1 Then
        sectionNum = 1
    Else
        sectionNum = 2
    End If

    If pageNo <= 2 Then

        ' Начало нужного раздела

        pos = doc.Sections(sectionNum).Range.Start
        Set rg = doc.Range(pos, pos)

    Else

        ' Следующая страница второго раздела

        pos = doc.Sections(2).Range.End - 1

        Set rg = doc.Range(pos, pos)

        ' Разрыв страницы

        rg.InsertBreak wdPageBreak

        ' Новое место для таблицы

        pos = doc.Sections(2).Range.End - 1

        Set rg = doc.Range(pos, pos)

    End If

    ' Создаём отдельную таблицу

    Set t = doc.Tables.Add(rg, rows.Count, 4)

    ' Разные размеры и координаты
    ' первого и последующих листов

    FormatPETable t, pageNo

    ' Заполнение

    For r = 1 To rows.Count

        data = rows(r)

        WritePERow t, r, data

    Next r

End Sub


' =====================================
' ОЧИСТКА РАЗДЕЛА
' =====================================

Sub ClearSectionBody(doc As Document, _
                     sectionNum As Long)

    Dim rg As Range

    Set rg = doc.Sections(sectionNum).Range.Duplicate

    ' Сохраняем разрыв раздела

    rg.End = rg.End - 1

    If rg.End > rg.Start Then
        rg.Delete
    End If

End Sub


' =====================================
' ПАРАМЕТРЫ СТРАНИЦЫ
' =====================================

Sub SetupPEPage(sec As Section)

    With sec.PageSetup

        .PaperSize = wdPaperA4
        .Orientation = wdOrientPortrait

        .TopMargin = 0
        .BottomMargin = 0
        .LeftMargin = 0
        .RightMargin = 0

    End With

End Sub


' =====================================
' ИЗМЕНЕНИЕ 5
' РАЗНЫЕ РАЗМЕРЫ ТАБЛИЦ ПО СТРАНИЦАМ
' =====================================

Sub FormatPETable(t As Table, _
                  Optional pageNo As Long = 1)

    Dim c As Cell

    t.AutoFitBehavior wdAutoFitFixed

    ' Размеры столбцов

    If pageNo = 1 Then

        t.Columns(1).Width = _
            CentimetersToPoints(1.99)

        t.Columns(2).Width = _
            CentimetersToPoints(11.02)

        t.Columns(3).Width = _
            CentimetersToPoints(1)

        t.Columns(4).Width = _
            CentimetersToPoints(4.2)

    Else

        t.Columns(1).Width = _
            CentimetersToPoints(2)

        t.Columns(2).Width = _
            CentimetersToPoints(11)

        t.Columns(3).Width = _
            CentimetersToPoints(1)

        t.Columns(4).Width = _
            CentimetersToPoints(4.5)

    End If

    ' =====================================
    ' ГРАНИЦЫ И ОТСТУПЫ
    ' =====================================

    With t

        .Borders.Enable = True

        .TopPadding = 0
        .BottomPadding = 0

        .LeftPadding = CentimetersToPoints(0.1)
        .RightPadding = CentimetersToPoints(0.2)

    End With

    ' =====================================
    ' ИЗМЕНЕНИЕ 6
    ' УБИРАЕМ ОТСТУПЫ В ПЕРВОЙ ГРАФЕ
    ' =====================================

    For Each c In t.Columns(1).Cells

        c.LeftPadding = 0
        c.RightPadding = 0

    Next c


    ' =====================================
    ' ВЫСОТА И ПОЛОЖЕНИЕ ТАБЛИЦЫ
    ' =====================================

    With t.Rows

        .HeightRule = wdRowHeightExactly
        .Height = CentimetersToPoints(0.8)

        .AllowBreakAcrossPages = False

        .WrapAroundText = True

        .RelativeHorizontalPosition = _
            wdRelativeHorizontalPositionPage

        .RelativeVerticalPosition = _
            wdRelativeVerticalPositionPage

        If pageNo = 1 Then

            ' Первый лист

            .HorizontalPosition = _
                CentimetersToPoints(2.12)

            .VerticalPosition = _
                CentimetersToPoints(2.13)

        Else

            ' Последующие листы

            .HorizontalPosition = _
                CentimetersToPoints(1.84)

            .VerticalPosition = _
                CentimetersToPoints(2.38)

        End If

        .AllowOverlap = False

    End With

    ' Вертикальное выравнивание

    t.Range.Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

    ' Шрифт

    With t.Range

        .Font.Name = "Arial"
        .Font.Size = 12

        With .ParagraphFormat

            .SpaceBefore = 0
            .SpaceAfter = 0

            .LeftIndent = 0
            .RightIndent = 0
            .FirstLineIndent = 0

            .LineSpacingRule = wdLineSpaceSingle

            .KeepWithNext = False
            .KeepTogether = False
            .WidowControl = False

        End With

    End With

End Sub


' =====================================
' ЗАПОЛНЕНИЕ ОДНОЙ СТРОКИ
' =====================================

Sub WritePERow(t As Table, _
               rowNum As Long, _
               data As Variant)

    Dim model As String
    Dim maker As String

    Dim tabPos As Single

    model = CStr(data(1))
    maker = CStr(data(2))

    ' Позиционное обозначение

    With t.Cell(rowNum, 1).Range

        .Text = CStr(data(0))

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With

    ' Количество

    With t.Cell(rowNum, 3).Range

        .Text = CStr(data(3))

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With

    ' Положение правой табуляции
    ' с учётом внутренних отступов

    tabPos = t.Columns(2).Width - _
             CentimetersToPoints(0.3)

    ' Наименование, изготовитель или ТУ

    With t.Cell(rowNum, 2).Range

        If model <> "" And maker <> "" Then

            .Text = model & vbTab & maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

            With .ParagraphFormat.TabStops

                .ClearAll

                .Add Position:=tabPos, _
                     Alignment:=wdAlignTabRight

            End With

        ElseIf maker <> "" Then

            .Text = maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphRight

        Else

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        End If

    End With

    ' Выравнивание по высоте

    t.Rows(rowNum).Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

End Sub


' =====================================
' ДОПОЛНЕНИЕ ПУСТЫМИ СТРОКАМИ
' =====================================

Sub FillPage(rows As Collection, _
             maxRows As Long)

    Do While rows.Count < maxRows

        rows.Add Array("", "", "", "")

    Loop

End Sub


' =====================================
' ПРОВЕРКА ШИРИНЫ ТЕКСТА
' =====================================

Function TextFits(t As Table, _
                  s As String, _
                  column As Long) As Boolean

    Dim rg As Range
    Dim count As Long

    If s = "" Then

        TextFits = True
        Exit Function

    End If

    t.Cell(1, column).Range.Text = s

    Set rg = t.Cell(1, column).Range.Duplicate

    rg.Font.Name = "Arial"
    rg.Font.Size = 12

    ' Исключаем маркер окончания ячейки

    rg.MoveEnd wdCharacter, -1

    rg.Document.Repaginate

    count = rg.ComputeStatistics(wdStatisticLines)

    TextFits = (count = 1)

End Function


' =====================================
' ПЕРЕНОС ОБЫЧНОГО ТЕКСТА
' =====================================

Function WrapText(t As Table, _
                  s As String, _
                  column As Long) As Collection

    Dim result As New Collection
    Dim words As Variant, word As Variant

    Dim line As String
    Dim candidate As String
    Dim part As String
    Dim n As Long

    s = Trim(Replace(s, vbTab, " "))

    Do While InStr(s, "  ") > 0

        s = Replace(s, "  ", " ")

    Loop

    If s = "" Then

        Set WrapText = result
        Exit Function

    End If

    words = Split(s, " ")

    For Each word In words

        part = CStr(word)

        If line = "" Then

            candidate = part

        Else

            candidate = line & " " & part

        End If

        If TextFits(t, candidate, column) Then

            line = candidate

        Else

            If line <> "" Then

                result.Add line
                line = ""

            End If

            If TextFits(t, part, column) Then

                line = part

            Else

                ' Слишком длинное слово

                Do While part <> ""

                    n = LongestPart(t, part, column)

                    result.Add Left(part, n)

                    part = Mid(part, n + 1)

                Loop

            End If

        End If

    Next word

    If line <> "" Then
        result.Add line
    End If

    Set WrapText = result

End Function


' =====================================
' ИЗМЕНЕНИЕ 7
' ПЕРЕНОС ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
'
' C101-C103 ->
' C101-
' C103
' =====================================

Function WrapReference(t As Table, _
                       s As String) As Collection

    Dim result As New Collection
    Dim pieces As Collection
    Dim items As Variant

    Dim item As String, line As String
    Dim candidate As String
    Dim leftPart As String, rightPart As String

    Dim i As Long, j As Long
    Dim p As Long

    If s = "" Then
        Set WrapReference = result
        Exit Function
    End If

    items = Split(s, ", ")

    For i = LBound(items) To UBound(items)

        item = CStr(items(i))

        ' Запятая относится к текущему
        ' обозначению, если оно не последнее

        If i < UBound(items) Then
            item = item & ","
        End If

        If line = "" Then
            candidate = item
        Else
            candidate = line & " " & item
        End If

        If TextFits(t, candidate, 1) Then

            line = candidate

        Else

            If line <> "" Then

                result.Add line
                line = ""

            End If

            ' Проверяем, помещается ли
            ' обозначение отдельно

            If TextFits(t, item, 1) Then

                line = item

            Else

                ' Переносим по дефису

                p = InStr(item, "-")

                If p > 0 Then

                    leftPart = Left(item, p)
                    rightPart = Mid(item, p + 1)

                    If TextFits(t, leftPart, 1) And _
                       TextFits(t, rightPart, 1) Then

                        result.Add leftPart
                        line = rightPart

                    Else

                        Set pieces = WrapText(t, item, 1)

                        For j = 1 To pieces.Count

                            If j = pieces.Count Then
                                line = CStr(pieces(j))
                            Else
                                result.Add CStr(pieces(j))
                            End If

                        Next j

                    End If

                Else

                    Set pieces = WrapText(t, item, 1)

                    For j = 1 To pieces.Count

                        If j = pieces.Count Then
                            line = CStr(pieces(j))
                        Else
                            result.Add CStr(pieces(j))
                        End If

                    Next j

                End If

            End If

        End If

    Next i

    If line <> "" Then
        result.Add line
    End If

    Set WrapReference = result

End Function


' =====================================
' МАКСИМАЛЬНАЯ ДЛИНА ФРАГМЕНТА
' =====================================

Function LongestPart(t As Table, _
                     s As String, _
                     column As Long) As Long

    Dim lo As Long, hi As Long
    Dim mid As Long

    lo = 1
    hi = Len(s)

    Do While lo < hi

        mid = (lo + hi + 1) \ 2

        If TextFits(t, Left(s, mid), column) Then

            lo = mid

        Else

            hi = mid - 1

        End If

    Loop

    LongestPart = lo

End Function


' =====================================
' РАЗДЕЛЕНИЕ НАИМЕНОВАНИЯ И ФИРМЫ/ТУ
' =====================================

Sub SplitName(s As String, _
              ByRef model As String, _
              ByRef maker As String)

    Dim p As Long
    Dim word As String

    s = Trim(s)

    model = s
    maker = ""

    ' Ищем обозначение фирмы

    p = InStr(1, s, " ф. ", vbTextCompare)

    If p > 0 Then

        model = Trim(Left(s, p - 1))
        maker = Trim(Mid(s, p + 1))

    Else

        ' Ищем ТУ в последнем слове

        p = InStrRev(s, " ")

        If p > 0 Then

            word = Trim(Mid(s, p + 1))

            If Len(word) > 2 And _
               Right(UCase(word), 2) = "ТУ" And _
               InStr(word, ".") > 0 Then

                model = Trim(Left(s, p - 1))
                maker = word

            End If

        End If

    End If

End Sub


' =====================================
' БУКВЕННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefCode(s As String) As String

    Dim i As Long

    s = UCase(Trim(s))

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then
            Exit For
        End If

    Next i

    RefCode = Left(s, i - 1)

End Function


' =====================================
' НОМЕР ЭЛЕМЕНТА
' =====================================

Function RefNum(s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function


' =====================================
' УДАЛЕНИЕ НАЗВАНИЯ ТИПА
'
' Только для группируемых элементов
' =====================================

Function RemoveType(s As String, _
                    code As String) As String

    Dim prefix As String

    Select Case code

        Case "C"
            prefix = "Конденсатор"

        Case "R"
            prefix = "Резистор"

        Case "DA", "DD", "U", "D"
            prefix = "Микросхема"

        Case Else
            prefix = ""

    End Select

    If prefix <> "" Then

        If StrComp(Left(s, Len(prefix) + 1), _
           prefix & " ", vbTextCompare) = 0 Then

            s = Mid(s, Len(prefix) + 2)

        End If

    End If

    RemoveType = s

End Function

End Function
