
Option Explicit

' =====================================
' СТРУКТУРЫ ДАННЫХ
' =====================================

Private Type PEItem
    Code As String
    Num As Long
    Model As String
    Maker As String
    TokenId As Long
End Type

Private Type PERecord
    Ref As String
    Code As String
    TypeKey As String
    Group As String
    Model As String
    Maker As String
    Qty As Long

    TUBlockId As Long
    TUHeaderSeries As String
End Type

Private Const FIRST_ROWS As Long = 23
Private Const NEXT_ROWS As Long = 30

Private Const REF_PADDING As Double = 0.08

' Минимум 4 записи с общим ТУ

Private Const MIN_TU_ITEMS As Long = 4


' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Sub MakeWordPE()

    Dim xl As Object, book As Object, sh As Object
    Dim doc As Document
    Dim testDoc1 As Document, testDoc2 As Document
    Dim test1 As Table, test2 As Table
    Dim probe As Table

    Dim a As Variant
    Dim el() As PEItem, tmp As PEItem
    Dim rec() As PERecord
    Dim parts As Variant, part As Variant

    Dim pages As Collection
    Dim pageRows As Collection
    Dim itemRows As Collection
    Dim firstRows As Collection
    Dim secondRows As Collection
    Dim tuRows As Collection
    Dim fullRows As Collection
    Dim piece As Variant

    Dim templateFile As String
    Dim excelFile As String
    Dim outputFile As String, baseName As String

    Dim ref As String, code As String
    Dim endCode As String
    Dim firstRef As String, lastRef As String
    Dim model As String, maker As String

    Dim total As Long, recCount As Long
    Dim last As Long, token As Long
    Dim i As Long, j As Long, k As Long
    Dim num As Long, n1 As Long, n2 As Long
    Dim pos As Long, runEnd As Long
    Dim limit As Long, gap As Long

    Dim needNormal As Long
    Dim needTU As Long
    Dim needExtra As Long
    Dim activeTU As Long
    Dim skipTU As Long
    Dim segmentStart As Long

    Dim showHeader As Boolean
    Dim initialTU As Boolean
    Dim useTU As Boolean

    On Error GoTo ErrorHandler


    ' =====================================
    ' ВЫБОР ШАБЛОНА WORD
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите шаблон Word"
        .Filters.Clear
        .Filters.Add "Word", "*.doc;*.docx;*.docm"

        If .Show <> -1 Then Exit Sub

        templateFile = .SelectedItems(1)

    End With


    ' =====================================
    ' ВЫБОР EXCEL
    ' =====================================

    With Application.FileDialog(3)

        .Title = "Выберите перечень Excel"
        .Filters.Clear
        .Filters.Add "Excel", "*.xlsx;*.xlsm;*.xls"

        If .Show <> -1 Then Exit Sub

        excelFile = .SelectedItems(1)

    End With


    ' =====================================
    ' ЧТЕНИЕ EXCEL
    ' =====================================

    Set xl = CreateObject("Excel.Application")

    Set book = xl.Workbooks.Open( _
        excelFile, 0, True)

    Set sh = book.Worksheets(1)

    last = sh.Cells(sh.Rows.Count, 1).End(-4162).Row

    If last < 2 Then

        MsgBox "В Excel нет данных."
        GoTo Finish

    End If

    a = sh.Range("A2:C" & last).Value2

    book.Close False
    Set book = Nothing

    xl.Quit
    Set xl = Nothing


    ' =====================================
    ' РАЗБОР ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
    ' =====================================

    total = 0
    token = 0

    For i = 1 To UBound(a, 1)

        SplitName CStr(a(i, 2)), model, maker

        ref = Trim(CStr(a(i, 1)))

        ref = Replace(ref, "–", "-")
        ref = Replace(ref, "—", "-")
        ref = Replace(ref, ",", " ")
        ref = Replace(ref, ";", " ")
        ref = Replace(ref, vbTab, " ")
        ref = Replace(ref, Chr(160), " ")

        Do While InStr(ref, "  ") > 0

            ref = Replace(ref, "  ", " ")

        Loop

        parts = Split(ref, " ")

        For Each part In parts

            ref = Trim(CStr(part))

            If ref <> "" Then

                pos = InStr(ref, "-")

                If pos > 0 Then

                    firstRef = Left(ref, pos - 1)
                    lastRef = Mid(ref, pos + 1)

                Else

                    firstRef = ref
                    lastRef = ref

                End If

                code = RefCode(firstRef)
                endCode = RefCode(lastRef)

                ' =================================
                ' ИЗМЕНЕНИЕ 1
                ' ПРОПУСК KT И КТ
                ' =================================

                If IsExcludedCode(code) Then

                    GoTo NextReference

                End If

                n1 = RefNum(firstRef)
                n2 = RefNum(lastRef)

                If code = "" Or n1 < 1 Or _
                   n2 < n1 Or _
                   (endCode <> "" And endCode <> code) Then

                    Err.Raise vbObjectError + 100, , _
                        "Ошибка обозначения: " & ref

                End If

                token = token + 1

                For num = n1 To n2

                    total = total + 1

                    ReDim Preserve el(1 To total)

                    el(total).Code = code
                    el(total).Num = num
                    el(total).Model = model
                    el(total).Maker = maker
                    el(total).TokenId = token

                Next num

            End If

NextReference:

        Next part

    Next i

    If total = 0 Then

        MsgBox "Нет элементов для перечня."
        GoTo Finish

    End If


    ' =====================================
    ' СОРТИРОВКА
    ' =====================================

    For i = 1 To total - 1

        For j = i + 1 To total

            If el(j).Code < el(i).Code Or _
               (el(j).Code = el(i).Code And _
                el(j).Num < el(i).Num) Then

                tmp = el(i)
                el(i) = el(j)
                el(j) = tmp

            End If

        Next j

    Next i


    ' =====================================
    ' ПРОВЕРКА ПОВТОРОВ
    ' =====================================

    For i = 2 To total

        If el(i).Code = el(i - 1).Code And _
           el(i).Num = el(i - 1).Num Then

            Err.Raise vbObjectError + 101, , _
                "Повтор обозначения: " & _
                el(i).Code & el(i).Num

        End If

    Next i


    ' =====================================
    ' СОЗДАНИЕ ЗАПИСЕЙ ПЕРЕЧНЯ
    ' =====================================

    recCount = 0
    i = 1

    Do While i <= total

        j = i

        Do While j < total

            If el(j + 1).Code <> el(i).Code Then
                Exit Do
            End If

            If el(j + 1).Num <> el(j).Num + 1 Then
                Exit Do
            End If

            If el(j + 1).Model <> el(i).Model Or _
               el(j + 1).Maker <> el(i).Maker Then
                Exit Do
            End If

            j = j + 1

        Loop

        recCount = recCount + 1
        ReDim Preserve rec(1 To recCount)

        rec(recCount).Ref = BuildReference(el, i, j)
        rec(recCount).Code = el(i).Code

        rec(recCount).TypeKey = _
            TypeKey(el(i).Code, el(i).Model)

        rec(recCount).Group = _
            GroupName(el(i).Code, el(i).Model)

        rec(recCount).Model = el(i).Model
        rec(recCount).Maker = el(i).Maker
        rec(recCount).Qty = j - i + 1

        i = j + 1

    Loop


    ' =====================================
    ' ИЗМЕНЕНИЕ 2
    ' ПОИСК ПОВТОРЯЮЩИХСЯ ТУ
    ' =====================================

    MarkRepeatedTU rec, recCount


    ' =====================================
    ' ПРОБНЫЕ ТАБЛИЦЫ
    ' =====================================

    Set testDoc1 = Documents.Add(Visible:=False)
    Set testDoc2 = Documents.Add(Visible:=False)

    SetupPEPage testDoc1.Sections(1)
    SetupPEPage testDoc2.Sections(1)

    Set test1 = testDoc1.Tables.Add( _
        testDoc1.Range(0, 0), 1, 4)

    Set test2 = testDoc2.Tables.Add( _
        testDoc2.Range(0, 0), 1, 4)

    FormatPETable test1, 1, True
    FormatPETable test2, 2, True


    ' =====================================
    ' РАСПРЕДЕЛЕНИЕ ПО СТРАНИЦАМ
    ' =====================================

    Set pages = New Collection
    Set pageRows = New Collection

    pages.Add pageRows

    limit = FIRST_ROWS
    i = 1

    Do While i <= recCount

        ' Определяем конец текущего типа

        runEnd = i

        Do While runEnd < recCount

            If rec(runEnd + 1).TypeKey <> _
               rec(i).TypeKey Then

                Exit Do

            End If

            runEnd = runEnd + 1

        Loop


        Do While i <= runEnd

            ' =================================
            ' ПРОВЕРКА ГРУППОВОГО ЗАГОЛОВКА
            ' =================================

            Do

                If pages.Count = 1 Then

                    Set probe = test1

                Else

                    Set probe = test2

                End If

                showHeader = False
                initialTU = False

                needNormal = 9999
                needTU = 9999

                If rec(i).Group <> "" And _
                   i < runEnd Then

                    ' Обычное оформление группы

                    Set firstRows = MakeItemRows( _
                        rec(i), False, probe)

                    Set secondRows = MakeItemRows( _
                        rec(i + 1), False, probe)

                    needNormal = 3 + _
                        firstRows.Count + _
                        secondRows.Count

                    ' Вариант с общим ТУ

                    If rec(i).TUBlockId > 0 And _
                       rec(i + 1).TUBlockId = _
                       rec(i).TUBlockId Then

                        Set tuRows = MakeTUHeaderRows( _
                            rec(i), probe)

                        Set firstRows = MakeItemRows( _
                            rec(i), False, probe, _
                            True, True)

                        Set secondRows = MakeItemRows( _
                            rec(i + 1), False, probe, _
                            True, True)

                        ' Пустая строка перед группой,
                        ' заголовок, строка ТУ,
                        ' минимум два компонента.
                        ' После заголовка пустой
                        ' строки здесь нет.

                        needTU = 2 + tuRows.Count + _
                            firstRows.Count + _
                            secondRows.Count

                    End If

                    ' Предпочитаем общее ТУ,
                    ' если оно помещается

                    If pageRows.Count + needTU <= limit Then

                        showHeader = True
                        initialTU = True

                        Exit Do

                    End If

                    If pageRows.Count + _
                       needNormal <= limit Then

                        showHeader = True

                        Exit Do

                    End If

                    ' Если группа не помещается,
                    ' пробуем начать её на
                    ' следующей странице

                    If pageRows.Count > 0 And _
                       (needTU <= NEXT_ROWS Or _
                        needNormal <= NEXT_ROWS) Then

                        NextPEPage pages, pageRows, limit

                        ' Повторяем проверку

                        GoTo RepeatGroupCheck

                    End If

                End If


                ' =================================
                ' ЕСЛИ ЗАГОЛОВКА ГРУППЫ НЕТ
                ' =================================

                Set fullRows = MakeItemRows( _
                    rec(i), True, probe)

                gap = 0

                If pageRows.Count > 0 Then
                    gap = 1
                End If

                If pageRows.Count + gap + _
                   fullRows.Count <= limit Then

                    Exit Do

                End If

                If pageRows.Count = 0 Then

                    Err.Raise vbObjectError + 102, , _
                        "Не помещается запись: " & _
                        rec(i).Ref

                End If

                NextPEPage pages, pageRows, limit

RepeatGroupCheck:

            Loop


            ' =================================
            ' ЗАГОЛОВОК ГРУППЫ
            ' =================================

            If showHeader Then

                AddBlank pageRows

                pageRows.Add Array( _
                    "", rec(i).Group, "", "", True)

                ' При наличии общего ТУ
                ' следующей строкой будет
                ' серия и ТУ, без пустой строки

                If Not initialTU Then
                    AddBlank pageRows
                End If

            Else

                ' Между типами элементов
                ' остаётся пустая строка

                If pageRows.Count > 0 Then
                    AddBlank pageRows
                End If

            End If


            ' =================================
            ' ЗАПОЛНЕНИЕ СТРАНИЦЫ
            ' =================================

            activeTU = 0
            skipTU = 0

            segmentStart = i

            ' Если общий ТУ не поместился
            ' непосредственно под группой,
            ' не создаём его посреди этого
            ' же блока после первого элемента

            If showHeader And Not initialTU Then

                skipTU = rec(i).TUBlockId

            End If

            Do While i <= runEnd

                useTU = False

                If showHeader And _
                   rec(i).TUBlockId > 0 Then

                    ' Уже открытый блок общего ТУ

                    If activeTU = rec(i).TUBlockId Then

                        useTU = True

                    ElseIf rec(i).TUBlockId <> skipTU Then

                        ' Проверяем, поместится ли
                        ' строка ТУ с минимум
                        ' двумя компонентами

                        If i < runEnd Then

                            If rec(i + 1).TUBlockId = _
                               rec(i).TUBlockId Then

                                Set tuRows = _
                                    MakeTUHeaderRows( _
                                    rec(i), probe)

                                Set firstRows = _
                                    MakeItemRows( _
                                    rec(i), False, probe, _
                                    True, True)

                                Set secondRows = _
                                    MakeItemRows( _
                                    rec(i + 1), False, probe, _
                                    True, True)

                                needExtra = 0

                                ' Между разными блоками ТУ
                                ' оставляем пустую строку,
                                ' но не сразу после
                                ' группового заголовка

                                If i > segmentStart Then
                                    needExtra = 1
                                End If

                                If pageRows.Count + _
                                   needExtra + _
                                   tuRows.Count + _
                                   firstRows.Count + _
                                   secondRows.Count <= limit Then

                                    If needExtra = 1 Then
                                        AddBlank pageRows
                                    End If

                                    For Each piece In tuRows
                                        pageRows.Add piece
                                    Next piece

                                    activeTU = _
                                        rec(i).TUBlockId

                                    useTU = True

                                End If

                            End If

                        End If

                    End If

                End If


                ' =================================
                ' ЗАПИСЬ ТЕКУЩЕГО КОМПОНЕНТА
                ' =================================

                If useTU Then

                    ' Общее ТУ уже наверху:
                    ' удаляем повтор ТУ и серии

                    Set itemRows = MakeItemRows( _
                        rec(i), False, probe, _
                        True, True)

                Else

                    ' Обычное оформление

                    Set itemRows = MakeItemRows( _
                        rec(i), Not showHeader, probe)

                End If

                If pageRows.Count + _
                   itemRows.Count > limit Then

                    Exit Do

                End If

                For Each piece In itemRows

                    pageRows.Add piece

                Next piece

                i = i + 1

            Loop


            ' Продолжение типа на новом листе

            If i <= runEnd Then

                NextPEPage pages, pageRows, limit

            End If

        Loop

    Loop


    ' =====================================
    ' ЗАВЕРШЕНИЕ ПОДГОТОВКИ СТРАНИЦ
    ' =====================================

    FillPage pageRows, limit

    testDoc1.Close wdDoNotSaveChanges
    testDoc2.Close wdDoNotSaveChanges

    Set testDoc1 = Nothing
    Set testDoc2 = Nothing


    ' =====================================
    ' ОТКРЫТИЕ WORD-ШАБЛОНА
    ' =====================================

    Set doc = Documents.Open( _
        FileName:=templateFile, _
        ReadOnly:=True, _
        AddToRecentFiles:=False)

    If doc.Sections.Count <> 3 Then

        Err.Raise vbObjectError + 103, , _
            "В шаблоне должно быть 3 раздела. " & _
            "Обнаружено: " & doc.Sections.Count

    End If

    baseName = Left(templateFile, _
        InStrRev(templateFile, ".") - 1)

    outputFile = baseName & "_PE.docx"

    k = 1

    Do While Dir(outputFile) <> ""

        outputFile = baseName & "_PE_" & _
                     k & ".docx"

        k = k + 1

    Loop

    ' Сохраняем копию шаблона

    doc.SaveAs2 FileName:=outputFile, _
                FileFormat:=wdFormatXMLDocument


    ' =====================================
    ' ПОДГОТОВКА РАЗДЕЛОВ
    ' =====================================

    ClearSectionBody doc, 2
    ClearSectionBody doc, 1

    doc.Sections(2).PageSetup.SectionStart = _
        wdSectionNewPage

    doc.Sections(3).PageSetup.SectionStart = _
        wdSectionNewPage

    SetupPEPage doc.Sections(1)
    SetupPEPage doc.Sections(2)

    ' Третий раздел не изменяем


    ' =====================================
    ' СОЗДАНИЕ ТАБЛИЦ
    ' =====================================

    For k = 1 To pages.Count

        Set pageRows = pages(k)

        WritePEPage doc, pageRows, k

    Next k

    doc.Repaginate
    doc.Save

    MsgBox "Перечень готов!" & vbCrLf & _
           "Страниц перечня: " & pages.Count & _
           vbCrLf & "Файл: " & outputFile


Finish:

    On Error Resume Next

    If Not book Is Nothing Then book.Close False
    If Not xl Is Nothing Then xl.Quit

    If Not testDoc1 Is Nothing Then
        testDoc1.Close wdDoNotSaveChanges
    End If

    If Not testDoc2 Is Nothing Then
        testDoc2.Close wdDoNotSaveChanges
    End If

    Exit Sub


ErrorHandler:

    MsgBox "Ошибка: " & Err.Description

    On Error Resume Next

    If Not doc Is Nothing Then
        doc.Close wdDoNotSaveChanges
    End If

    Resume Finish

End Sub


' =====================================
' ИСКЛЮЧЕНИЕ KT И КТ
' =====================================

Function IsExcludedCode(code As String) As Boolean

    Dim s As String

    s = UCase(Trim(code))

    ' Приводим кириллицу к латинице
    ' для проверки именно этого кода

    s = Replace(s, "К", "K")
    s = Replace(s, "Т", "T")

    IsExcludedCode = (s = "KT")

End Function


' =====================================
' ПОИСК ОБЩИХ ТУ
' =====================================

Sub MarkRepeatedTU(rec() As PERecord, _
                   recCount As Long)

    Dim i As Long, j As Long, k As Long
    Dim blockId As Long

    Dim series As String
    Dim otherSeries As String

    i = 1
    blockId = 0

    Do While i <= recCount

        ' Только группируемые элементы
        ' и именно технические условия

        If rec(i).Group <> "" And _
           IsTU(rec(i).Maker) Then

            j = i

            ' Поиск непрерывной
            ' последовательности с общим ТУ

            Do While j < recCount

                If rec(j + 1).TypeKey <> _
                   rec(i).TypeKey Then
                    Exit Do
                End If

                If rec(j + 1).Group <> _
                   rec(i).Group Then
                    Exit Do
                End If

                If StrComp(rec(j + 1).Maker, _
                           rec(i).Maker, _
                           vbTextCompare) <> 0 Then
                    Exit Do
                End If

                j = j + 1

            Loop

            ' Не менее 4 записей

            If j - i + 1 >= MIN_TU_ITEMS Then

                blockId = blockId + 1

                ' Определяем общую серию

                series = SeriesCode( _
                    rec(i).Model, rec(i).Code)

                For k = i + 1 To j

                    otherSeries = SeriesCode( _
                        rec(k).Model, rec(k).Code)

                    If StrComp(series, otherSeries, _
                               vbTextCompare) <> 0 Then

                        ' Серии различаются.
                        ' Выносим только ТУ.

                        series = ""
                        Exit For

                    End If

                Next k

                ' Помечаем записи блока

                For k = i To j

                    rec(k).TUBlockId = blockId

                    rec(k).TUHeaderSeries = series

                Next k

            End If

            i = j + 1

        Else

            i = i + 1

        End If

    Loop

End Sub


' =====================================
' ПРОВЕРКА, ЧТО СТРОКА ЯВЛЯЕТСЯ ТУ
' =====================================

Function IsTU(s As String) As Boolean

    s = UCase(Trim(s))

    IsTU = False

    If Len(s) < 3 Then Exit Function

    If Right(s, 2) = "ТУ" And _
       InStr(s, ".") > 0 Then

        IsTU = True

    End If

End Function


' =====================================
' ОПРЕДЕЛЕНИЕ СЕРИИ КОМПОНЕНТА
'
' Например:
' Конденсатор К10-79 0,1 мкФ
' -> К10-79
' =====================================

Function SeriesCode(model As String, _
                    code As String) As String

    Dim s As String
    Dim word As String

    Dim p As Long, i As Long
    Dim hasDigit As Boolean

    ' Убираем общий тип компонента

    s = RemoveType(model, code)
    s = Trim(s)

    p = InStr(s, " ")

    If p <= 1 Then Exit Function

    word = Left(s, p - 1)

    ' Серия должна содержать дефис
    ' и хотя бы одну цифру

    If InStr(word, "-") = 0 Then
        Exit Function
    End If

    hasDigit = False

    For i = 1 To Len(word)

        If Mid(word, i, 1) Like "#" Then

            hasDigit = True
            Exit For

        End If

    Next i

    If hasDigit Then
        SeriesCode = word
    End If

End Function


' =====================================
' СТРОКА ОБЩЕГО ТУ
'
' К10-79 ШКАБ.123456.001ТУ
' =====================================

Function MakeTUHeaderRows(item As PERecord, _
                          t As Table) As Collection

    Dim result As New Collection
    Dim lines As Collection

    Dim s As String
    Dim piece As Variant

    s = Trim(item.TUHeaderSeries)

    If s <> "" Then
        s = s & " "
    End If

    s = s & item.Maker

    Set lines = WrapText(t, s, 2)

    For Each piece In lines

        ' Последний параметр обозначает
        ' строку общего ТУ

        result.Add Array( _
            "", CStr(piece), "", "", _
            False, wdAlignParagraphCenter, _
            False, True)

    Next piece

    Set MakeTUHeaderRows = result

End Function


' =====================================
' ФОРМИРОВАНИЕ СТРОК КОМПОНЕНТА
' =====================================

Function MakeItemRows(item As PERecord, _
                      fullType As Boolean, _
                      t As Table, _
                      Optional hideTU As Boolean = False, _
                      Optional hideSeries As Boolean = False) _
                      As Collection

    Dim result As New Collection
    Dim chunks As New Collection

    Dim textLines As Collection
    Dim refLines As Collection

    Dim model As String, maker As String
    Dim pair As Variant, piece As Variant

    Dim i As Long, n As Long
    Dim refAlign As Long
    Dim refText As String, qtyText As String
    Dim hasRefWrap As Boolean

    model = item.Model
    maker = item.Maker


    ' =====================================
    ' ОБРАБОТКА НАЗВАНИЯ
    ' =====================================

    If item.Group <> "" Then

        If fullType Then

            model = FullTypeName(model, item.Code)

        Else

            model = RemoveType(model, item.Code)

        End If

    End If


    ' =====================================
    ' УДАЛЕНИЕ ВЫНЕСЕННОГО ТУ И СЕРИИ
    ' =====================================

    If hideTU Then

        maker = ""

    End If

    If hideSeries And _
       item.TUHeaderSeries <> "" Then

        model = RemoveLeadingSeries( _
            model, item.TUHeaderSeries)

    End If


    ' =====================================
    ' ПОЗИЦИОННОЕ ОБОЗНАЧЕНИЕ
    ' =====================================

    Set refLines = WrapReference(t, item.Ref)

    hasRefWrap = (refLines.Count > 1)


    ' =====================================
    ' ФОРМИРОВАНИЕ НАИМЕНОВАНИЯ
    ' =====================================

    If hasRefWrap Then

        ' При переносе обозначения
        ' фирма и ТУ идут ниже модели

        Set textLines = WrapText(t, model, 2)

        For Each piece In textLines

            chunks.Add Array(CStr(piece), "")

        Next piece

        Set textLines = WrapText(t, maker, 2)

        For Each piece In textLines

            chunks.Add Array("", CStr(piece))

        Next piece

    Else

        ' Если модель и изготовитель
        ' помещаются вместе

        If model <> "" And maker <> "" And _
           TextFits(t, model & Space(4) & maker, 2) Then

            chunks.Add Array(model, maker)

        Else

            Set textLines = WrapText(t, model, 2)

            For Each piece In textLines

                chunks.Add Array(CStr(piece), "")

            Next piece

            Set textLines = WrapText(t, maker, 2)

            For Each piece In textLines

                chunks.Add Array("", CStr(piece))

            Next piece

        End If

    End If

    If chunks.Count = 0 Then

        chunks.Add Array("", "")

    End If

    n = chunks.Count

    If refLines.Count > n Then
        n = refLines.Count
    End If


    ' =====================================
    ' ЗАПОЛНЕНИЕ ФИЗИЧЕСКИХ СТРОК
    ' =====================================

    For i = 1 To n

        If i <= chunks.Count Then

            pair = chunks(i)

        Else

            pair = Array("", "")

        End If

        refText = ""
        qtyText = ""

        If i <= refLines.Count Then
            refText = CStr(refLines(i))
        End If

        ' Выравнивание обозначения

        refAlign = wdAlignParagraphCenter

        If hasRefWrap Then

            If i = 1 Then

                refAlign = wdAlignParagraphLeft

            ElseIf i = refLines.Count Then

                refAlign = wdAlignParagraphRight

            End If

        End If

        ' Количество на последней строке

        If i = n Then
            qtyText = CStr(item.Qty)
        End If

        result.Add Array( _
            refText, _
            CStr(pair(0)), _
            CStr(pair(1)), _
            qtyText, _
            False, _
            refAlign, _
            (hasRefWrap And refText <> ""), _
            False)

    Next i

    Set MakeItemRows = result

End Function


' =====================================
' УДАЛЕНИЕ ОБЩЕЙ СЕРИИ ИЗ МОДЕЛИ
' =====================================

Function RemoveLeadingSeries(s As String, _
                             series As String) As String

    Dim tail As String

    s = Trim(s)

    If StartsWithTerm(s, series) Then

        tail = Trim(Mid(s, Len(series) + 1))

        ' Не оставляем модель пустой

        If tail <> "" Then
            s = tail
        End If

    End If

    RemoveLeadingSeries = s

End Function

' =====================================
' СОСТАВЛЕНИЕ ОБОЗНАЧЕНИЯ
' =====================================

Function BuildReference(el() As PEItem, _
                        first As Long, _
                        last As Long) As String

    Dim k As Long, j As Long
    Dim result As String, part As String

    result = ""
    k = first

    Do While k <= last

        j = k

        Do While j < last

            If el(j + 1).TokenId <> _
               el(k).TokenId Then
                Exit Do
            End If

            If el(j + 1).Num <> _
               el(j).Num + 1 Then
                Exit Do
            End If

            j = j + 1

        Loop

        If j = k Then

            part = el(k).Code & el(k).Num

        Else

            part = el(k).Code & el(k).Num & _
                   "-" & el(j).Code & el(j).Num

        End If

        ' Без пробела после запятой

        If result <> "" Then
            result = result & ","
        End If

        result = result & part

        k = j + 1

    Loop

    BuildReference = result

End Function

' =====================================
' ПРОВЕРКА НАЧАЛА НАЗВАНИЯ
' =====================================

Function StartsWithTerm(s As String, _
                        term As String) As Boolean

    Dim nextChar As String

    s = Trim(s)

    If term = "" Then Exit Function
    If Len(s) < Len(term) Then Exit Function

    If StrComp(Left(s, Len(term)), _
               term, vbTextCompare) <> 0 Then

        Exit Function

    End If

    If Len(s) = Len(term) Then

        StartsWithTerm = True
        Exit Function

    End If

    nextChar = Mid(s, Len(term) + 1, 1)

    StartsWithTerm = _
        (nextChar = " " Or _
         nextChar = "-" Or _
         nextChar = "," Or _
         nextChar = "(")

End Function

' =====================================

Sub WritePEPage(doc As Document, _
                rows As Collection, _
                pageNo As Long)

    Dim rg As Range
    Dim t As Table
    Dim data As Variant

    Dim r As Long, pos As Long
    Dim sectionNum As Long

    If pageNo = 1 Then
        sectionNum = 1
    Else
        sectionNum = 2
    End If

    If pageNo <= 2 Then

        pos = doc.Sections(sectionNum).Range.Start

        Set rg = doc.Range(pos, pos)

    Else

        pos = doc.Sections(2).Range.End - 1

        Set rg = doc.Range(pos, pos)

        rg.InsertBreak wdPageBreak

        pos = doc.Sections(2).Range.End - 1

        Set rg = doc.Range(pos, pos)

    End If

    Set t = doc.Tables.Add(rg, rows.Count, 4)

    FormatPETable t, pageNo

    For r = 1 To rows.Count

        data = rows(r)

        WritePERow t, r, data

    Next r

End Sub

' =====================================
' ОФОРМЛЕНИЕ ТАБЛИЦ
' =====================================

Sub FormatPETable(t As Table, _
                  Optional pageNo As Long = 1, _
                  Optional forMeasure As Boolean = False)

    Dim c As Cell

    t.AutoFitBehavior wdAutoFitFixed


    ' =====================================
    ' ШИРИНЫ СТОЛБЦОВ
    ' =====================================

    If pageNo = 1 Then

        t.Columns(1).Width = _
            CentimetersToPoints(1.99)

        t.Columns(2).Width = _
            CentimetersToPoints(11.02)

        t.Columns(3).Width = _
            CentimetersToPoints(1)

        t.Columns(4).Width = _
            CentimetersToPoints(4.2)

    Else

        t.Columns(1).Width = _
            CentimetersToPoints(2)

        t.Columns(2).Width = _
            CentimetersToPoints(11)

        t.Columns(3).Width = _
            CentimetersToPoints(1)

        t.Columns(4).Width = _
            CentimetersToPoints(4.5)

    End If


    ' =====================================
    ' ГРАНИЦЫ И ОТСТУПЫ
    ' =====================================

    With t

        .Borders.Enable = True

        .TopPadding = 0
        .BottomPadding = 0

        .LeftPadding = CentimetersToPoints(0.1)
        .RightPadding = CentimetersToPoints(0.2)

    End With

    ' Отступы первой графы

    For Each c In t.Columns(1).Cells

        If forMeasure Then

            c.LeftPadding = _
                CentimetersToPoints(REF_PADDING)

            c.RightPadding = _
                CentimetersToPoints(REF_PADDING)

        Else

            c.LeftPadding = 0
            c.RightPadding = 0

        End If

    Next c


    ' =====================================
    ' ВЫСОТА И ПОЛОЖЕНИЕ ТАБЛИЦЫ
    ' =====================================

    With t.Rows

        .AllowBreakAcrossPages = False

        If forMeasure Then

            .HeightRule = wdRowHeightAtLeast

        Else

            .HeightRule = wdRowHeightExactly
            .Height = CentimetersToPoints(0.8)

            .WrapAroundText = True

            .RelativeHorizontalPosition = _
                wdRelativeHorizontalPositionPage

            .RelativeVerticalPosition = _
                wdRelativeVerticalPositionPage

            If pageNo = 1 Then

                .HorizontalPosition = _
                    CentimetersToPoints(2.12)

                .VerticalPosition = _
                    CentimetersToPoints(2.13)

            Else

                .HorizontalPosition = _
                    CentimetersToPoints(1.84)

                .VerticalPosition = _
                    CentimetersToPoints(2.38)

            End If

            .AllowOverlap = False

        End If

    End With


    ' =====================================
    ' ШРИФТ И ВЫРАВНИВАНИЕ
    ' =====================================

    t.Range.Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

    With t.Range

        .Font.Name = "Arial"
        .Font.Size = 12

        With .ParagraphFormat

            .SpaceBefore = 0
            .SpaceAfter = 0

            .LeftIndent = 0
            .RightIndent = 0
            .FirstLineIndent = 0

            .LineSpacingRule = wdLineSpaceSingle

            .KeepWithNext = False
            .KeepTogether = False
            .WidowControl = False

        End With

    End With

End Sub


' =====================================
' ЗАПОЛНЕНИЕ СТРОКИ
' =====================================

Sub WritePERow(t As Table, _
               rowNum As Long, _
               data As Variant)

    Dim model As String, maker As String

    Dim isHeader As Boolean
    Dim isTUHeader As Boolean

    Dim refAlign As Long
    Dim refWrapped As Boolean

    Dim tabPos As Single
    Dim c As Cell

    model = CStr(data(1))
    maker = CStr(data(2))
    isHeader = CBool(data(4))

    refAlign = wdAlignParagraphCenter
    refWrapped = False
    isTUHeader = False

    If UBound(data) >= 5 Then
        refAlign = CLng(data(5))
    End If

    If UBound(data) >= 6 Then
        refWrapped = CBool(data(6))
    End If

    If UBound(data) >= 7 Then
        isTUHeader = CBool(data(7))
    End If


    ' =====================================
    ' ПОЗИЦИОННОЕ ОБОЗНАЧЕНИЕ
    ' =====================================

    Set c = t.Cell(rowNum, 1)

    If refWrapped Then

        c.LeftPadding = _
            CentimetersToPoints(REF_PADDING)

        c.RightPadding = _
            CentimetersToPoints(REF_PADDING)

    Else

        c.LeftPadding = 0
        c.RightPadding = 0

    End If

    With c.Range

        .Text = CStr(data(0))
        .ParagraphFormat.Alignment = refAlign

    End With


    ' =====================================
    ' КОЛИЧЕСТВО
    ' =====================================

    With t.Cell(rowNum, 3).Range

        .Text = CStr(data(3))

        .ParagraphFormat.Alignment = _
            wdAlignParagraphCenter

    End With


    ' =====================================
    ' НАИМЕНОВАНИЕ
    ' =====================================

    Set c = t.Cell(rowNum, 2)

    tabPos = t.Columns(2).Width - _
             CentimetersToPoints(0.3)

    If isTUHeader Then

        ' ИЗМЕНЕНИЕ:
        ' В строке общего ТУ нет
        ' внутренних отступов

        c.LeftPadding = 0
        c.RightPadding = 0

    End If

    With c.Range

        If isHeader Then

            ' Заголовок группы по центру

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphCenter

        ElseIf isTUHeader Then

            ' Серия и ТУ рядом,
            ' с обычным пробелом,
            ' без правой табуляции

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        ElseIf model <> "" And maker <> "" Then

            .Text = model & vbTab & maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

            With .ParagraphFormat.TabStops

                .ClearAll

                .Add Position:=tabPos, _
                     Alignment:=wdAlignTabRight

            End With

        ElseIf maker <> "" Then

            .Text = maker

            .ParagraphFormat.Alignment = _
                wdAlignParagraphRight

        Else

            .Text = model

            .ParagraphFormat.Alignment = _
                wdAlignParagraphLeft

        End If

    End With

    t.Rows(rowNum).Cells.VerticalAlignment = _
        wdCellAlignVerticalCenter

End Sub


' =====================================
' ПРОВЕРКА ШИРИНЫ ТЕКСТА
' =====================================

Function TextFits(t As Table, _
                  s As String, _
                  column As Long) As Boolean

    Dim rg As Range
    Dim count As Long

    If s = "" Then

        TextFits = True
        Exit Function

    End If

    t.Cell(1, column).Range.Text = s

    Set rg = t.Cell(1, column).Range.Duplicate

    rg.Font.Name = "Arial"
    rg.Font.Size = 12

    rg.End = rg.End - 2

    rg.Document.Repaginate

    count = rg.ComputeStatistics(wdStatisticLines)

    TextFits = (count = 1)

End Function


' =====================================
' ПЕРЕНОС ОБЫЧНОГО ТЕКСТА
' =====================================

Function WrapText(t As Table, _
                  s As String, _
                  column As Long) As Collection

    Dim result As New Collection
    Dim words As Variant, word As Variant

    Dim line As String
    Dim candidate As String
    Dim part As String
    Dim n As Long

    s = Trim(Replace(s, vbTab, " "))

    Do While InStr(s, "  ") > 0

        s = Replace(s, "  ", " ")

    Loop

    If s = "" Then

        Set WrapText = result
        Exit Function

    End If

    words = Split(s, " ")

    For Each word In words

        part = CStr(word)

        If line = "" Then

            candidate = part

        Else

            candidate = line & " " & part

        End If

        If TextFits(t, candidate, column) Then

            line = candidate

        Else

            If line <> "" Then

                result.Add line
                line = ""

            End If

            If TextFits(t, part, column) Then

                line = part

            Else

                Do While part <> ""

                    n = LongestPart(t, part, column)

                    If n < 1 Then

                        Err.Raise vbObjectError + 110, , _
                            "Ошибка переноса: " & part

                    End If

                    result.Add Left(part, n)

                    part = Mid(part, n + 1)

                Loop

            End If

        End If

    Next word

    If line <> "" Then
        result.Add line
    End If

    Set WrapText = result

End Function


' =====================================
' ПЕРЕНОС ПОЗИЦИОННЫХ ОБОЗНАЧЕНИЙ
' =====================================

Function WrapReference(t As Table, _
                       s As String) As Collection

    Dim result As New Collection
    Dim items As Variant

    Dim item As String
    Dim suffix As String
    Dim line As String
    Dim candidate As String

    Dim leftPart As String
    Dim rightPart As String

    Dim i As Long, p As Long
    Dim firstNum As Long, lastNum As Long
    Dim previousNum As Long

    Dim forceRange As Boolean
    Dim forceNewLine As Boolean

    If s = "" Then

        Set WrapReference = result
        Exit Function

    End If

    items = Split(s, ",")

    previousNum = 0
    line = ""

    For i = LBound(items) To UBound(items)

        item = Trim(CStr(items(i)))
        suffix = ""

        If i < UBound(items) Then
            suffix = ","
        End If

        p = InStr(item, "-")

        If p > 0 Then

            leftPart = Left(item, p)
            rightPart = Mid(item, p + 1)

            firstNum = RefNum(Left(item, p - 1))
            lastNum = RefNum(rightPart)

        Else

            firstNum = RefNum(item)
            lastNum = firstNum

        End If


        ' =====================================
        ' ПРИНУДИТЕЛЬНЫЙ ПЕРЕНОС ДИАПАЗОНА
        ' =====================================

        forceRange = False

        If p > 0 Then

            If firstNum > 100 Or lastNum > 100 Then
                forceRange = True
            End If

        End If

        If p > 0 And _
           (forceRange Or _
            Not TextFits(t, item & suffix, 1)) Then

            If line <> "" Then

                result.Add line
                line = ""

            End If

            If Not TextFits(t, leftPart, 1) Then

                Err.Raise vbObjectError + 111, , _
                    "Не помещается: " & leftPart

            End If

            result.Add leftPart

            candidate = rightPart & suffix

            If Not TextFits(t, candidate, 1) Then

                Err.Raise vbObjectError + 112, , _
                    "Не помещается: " & candidate

            End If

            line = candidate

        Else

            candidate = item & suffix
            forceNewLine = False

            If line <> "" Then

                If previousNum > 100 Or _
                   firstNum > 100 Then

                    forceNewLine = True

                End If

            End If

            If line = "" Then

                If Not TextFits(t, candidate, 1) Then

                    Err.Raise vbObjectError + 113, , _
                        "Не помещается: " & candidate

                End If

                line = candidate

            ElseIf Not forceNewLine And _
                   TextFits(t, line & candidate, 1) Then

                line = line & candidate

            Else

                result.Add line

                If Not TextFits(t, candidate, 1) Then

                    Err.Raise vbObjectError + 114, , _
                        "Не помещается: " & candidate

                End If

                line = candidate

            End If

        End If

        previousNum = lastNum

    Next i

    If line <> "" Then
        result.Add line
    End If

    Set WrapReference = result

End Function


' =====================================
' МАКСИМАЛЬНАЯ ДЛИНА ФРАГМЕНТА
' =====================================

Function LongestPart(t As Table, _
                     s As String, _
                     column As Long) As Long

    Dim lo As Long, hi As Long
    Dim mid As Long

    lo = 1
    hi = Len(s)

    Do While lo < hi

        mid = (lo + hi + 1) \ 2

        If TextFits(t, Left(s, mid), column) Then

            lo = mid

        Else

            hi = mid - 1

        End If

    Loop

    LongestPart = lo

End Function


' =====================================
' РАЗДЕЛЕНИЕ МОДЕЛИ И ФИРМЫ / ТУ
' =====================================

Sub SplitName(s As String, _
              ByRef model As String, _
              ByRef maker As String)

    Dim p As Long, word As String

    s = Trim(s)

    model = s
    maker = ""

    p = InStr(1, s, " ф. ", vbTextCompare)

    If p > 0 Then

        model = Trim(Left(s, p - 1))
        maker = Trim(Mid(s, p + 1))

    Else

        p = InStrRev(s, " ")

        If p > 0 Then

            word = Trim(Mid(s, p + 1))

            If IsTU(word) Then

                model = Trim(Left(s, p - 1))
                maker = word

            End If

        End If

    End If

    ' Нормализация названий

    model = NormalizeModelName(model)

End Sub


' =====================================
' ИЗМЕНЕНИЕ 3
' НОРМАЛИЗАЦИЯ НАИМЕНОВАНИЙ
'
' НОВЫЕ ЗАМЕНЫ ДОБАВЛЯТЬ ЗДЕСЬ
' =====================================

Function NormalizeModelName(s As String) As String

    s = Trim(s)

    ' Возвращаем "Модуль питания"

    s = ReplaceLeading(s, _
        "Модуль электропитания", _
        "Модуль питания")

    s = ReplaceLeading(s, _
        "Источник электропитания", _
        "Модуль питания")

    s = ReplaceLeading(s, _
        "Источник питания", _
        "Модуль питания")

    ' Пример новой замены:
    '
    ' s = ReplaceLeading(s, _
    '     "Старое название", _
    '     "Новое название")

    NormalizeModelName = s

End Function


' =====================================
' ЗАМЕНА НАЧАЛА НАИМЕНОВАНИЯ
' =====================================

Function ReplaceLeading(s As String, _
                        oldText As String, _
                        newText As String) As String

    s = Trim(s)

    If StartsWithTerm(s, oldText) Then

        ReplaceLeading = newText & _
            Mid(s, Len(oldText) + 1)

    Else

        ReplaceLeading = s

    End If

End Function


' =====================================
' БУКВЕННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefCode(s As String) As String

    Dim i As Long

    s = UCase(Trim(s))

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then
            Exit For
        End If

    Next i

    RefCode = Left(s, i - 1)

End Function


' =====================================
' ЧИСЛОВОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefNum(s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function

Function PrefixName(code As String, _
                    model As String) As String

    Select Case code

        Case "C"

            PrefixName = "Конденсатор"

        Case "R"

            PrefixName = "Резистор"

        Case "D", "DA", "DD", "U"

            PrefixName = "Микросхема"

        Case "G"

            If GComponentKind(model) = "POWER" Then

                PrefixName = "Модуль питания"

            Else

                PrefixName = ""

            End If

        Case Else

            PrefixName = ""

    End Select

End Function


' =====================================
' УДАЛЕНИЕ НАЗВАНИЯ ТИПА
' =====================================

Function RemoveType(s As String, _
                    code As String) As String

    Dim prefix As String

    s = Trim(s)
    prefix = PrefixName(code, s)

    If prefix <> "" Then

        If StartsWithTerm(s, prefix) Then

            s = Trim(Mid(s, Len(prefix) + 1))

        End If

    End If

    RemoveType = s

End Function


' =====================================
' ПОЛНОЕ НАЗВАНИЕ БЕЗ ГРУППЫ
' =====================================

Function FullTypeName(s As String, _
                      code As String) As String

    Dim prefix As String

    s = Trim(s)
    prefix = PrefixName(code, s)

    If prefix = "" Then

        FullTypeName = s
        Exit Function

    End If

    ' Модули питания и генераторы
    ' не получают лишних приставок

    If StartsWithTerm(s, "Модуль питания") Or _
       StartsWithTerm(s, "Генератор") Then

        FullTypeName = s
        Exit Function

    End If

    If StartsWithTerm(s, prefix) Then

        FullTypeName = s

    Else

        FullTypeName = prefix & " " & s

    End If

End Function

End Function


' =====================================
' КЛАССИФИКАЦИЯ КОМПОНЕНТОВ G
' =====================================

Function GComponentKind(model As String) As String

    If StartsWithTerm(model, _
                      "Модуль питания") Then

        GComponentKind = "POWER"

    ElseIf StartsWithTerm(model, _
                          "Генератор") Then

        GComponentKind = "GENERATOR"

    Else

        GComponentKind = "OTHER"

    End If

End Function


' =====================================
' ПРОВЕРКА НАЧАЛА НАЗВАНИЯ
' =====================================

Function StartsWithTerm(s As String, _
                        term As String) As Boolean

    Dim nextChar As String

    s = Trim(s)

    If term = "" Then Exit Function
    If Len(s) < Len(term) Then Exit Function

    If StrComp(Left(s, Len(term)), _
               term, vbTextCompare) <> 0 Then

        Exit Function

    End If

    If Len(s) = Len(term) Then

        StartsWithTerm = True
        Exit Function

    End If

    nextChar = Mid(s, Len(term) + 1, 1)

    StartsWithTerm = _
        (nextChar = " " Or _
         nextChar = "-" Or _
         nextChar = "," Or _
         nextChar = "(")

End Function


' =====================================
' НАЗВАНИЯ ГРУПП
' =====================================

Function GroupName(code As String, _
                   model As String) As String

    Select Case code

        Case "C"

            GroupName = "КОНДЕНСАТОРЫ"

        Case "R"

            GroupName = "РЕЗИСТОРЫ"

        Case "D", "DA", "DD", "U"

            GroupName = "МИКРОСХЕМЫ"

        Case "G"

            If GComponentKind(model) = "POWER" Then

                GroupName = "МОДУЛИ ПИТАНИЯ"

            Else

                GroupName = ""

            End If

        Case Else

            GroupName = ""

    End Select

End Function
