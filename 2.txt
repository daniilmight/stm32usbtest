
Option Explicit

' Данные одного элемента
' ИЗМЕНЕНИЕ 1: храним каждый номер отдельно

Private Type PEItem
    Code As String
    Num As Long
    Model As String
    Maker As String
End Type


' =====================================
' ОСНОВНОЙ МАКРОС
' =====================================

Sub MakePE()

    Dim src As Worksheet, dst As Worksheet
    Dim wb As Workbook
    Dim a As Variant

    Dim el() As PEItem
    Dim temp As PEItem

    Dim i As Long, j As Long, k As Long
    Dim total As Long, last As Long
    Dim page As Long, r As Long, maxRow As Long
    Dim n As Long, num As Long
    Dim num1 As Long, num2 As Long
    Dim p As Long

    Dim parts As Variant, part As Variant
    Dim ref As String, code As String
    Dim firstRef As String, lastRef As String
    Dim model As String, maker As String
    Dim group As String, oldGroup As String

    Dim names As Collection
    Dim makers As Collection

    Set src = ActiveSheet

    last = src.Cells(src.Rows.Count, 1).End(xlUp).Row
    If last < 2 Then Exit Sub

    a = src.Range("A2:C" & last).Value


    ' =====================================
    ' ИЗМЕНЕНИЕ 2
    ' Разбор диапазонов на отдельные номера
    ' =====================================

    total = 0

    For i = 1 To UBound(a, 1)

        SplitName CStr(a(i, 2)), model, maker

        ref = Trim(CStr(a(i, 1)))

        ref = Replace(ref, "–", "-")
        ref = Replace(ref, "—", "-")
        ref = Replace(ref, ";", ",")

        parts = Split(ref, ",")

        For Each part In parts

            ref = Trim(CStr(part))

            If ref <> "" Then

                p = InStr(ref, "-")

                If p > 0 Then
                    firstRef = Trim(Left(ref, p - 1))
                    lastRef = Trim(Mid(ref, p + 1))
                Else
                    firstRef = ref
                    lastRef = ref
                End If

                code = RefCode(firstRef)
                num1 = RefNum(firstRef)
                num2 = RefNum(lastRef)

                If code = "" Or num1 = 0 Or _
                   num2 < num1 Then

                    MsgBox "Ошибка обозначения: " & ref
                    Exit Sub

                End If

                For num = num1 To num2

                    total = total + 1
                    ReDim Preserve el(1 To total)

                    el(total).Code = code
                    el(total).Num = num
                    el(total).Model = model
                    el(total).Maker = maker

                Next num

            End If

        Next part

    Next i

    If total = 0 Then Exit Sub


    ' =====================================
    ' ИЗМЕНЕНИЕ 3
    ' Сортировка каждого отдельного номера
    ' =====================================

    For i = 1 To total - 1

        For j = i + 1 To total

            If el(j).Code < el(i).Code Or _
               (el(j).Code = el(i).Code And _
                el(j).Num < el(i).Num) Then

                temp = el(i)
                el(i) = el(j)
                el(j) = temp

            End If

        Next j

    Next i


    ' Создание новой книги

    Set wb = Workbooks.Add(xlWBATWorksheet)
    Set dst = wb.Worksheets(1)

    page = 1
    r = 2
    maxRow = 24
    oldGroup = ""

    PreparePage dst, page


    ' =====================================
    ' ИЗМЕНЕНИЕ 4
    ' Объединение только соседних
    ' одинаковых элементов
    ' =====================================

    i = 1

    Do While i <= total

        j = i

        Do While j < total

            If el(j + 1).Code <> el(i).Code Then
                Exit Do
            End If

            If el(j + 1).Num <> el(j).Num + 1 Then
                Exit Do
            End If

            If el(j + 1).Model <> el(i).Model Or _
               el(j + 1).Maker <> el(i).Maker Then
                Exit Do
            End If

            j = j + 1

        Loop

        code = el(i).Code

        If i = j Then
            ref = code & el(i).Num
        Else
            ref = code & el(i).Num & "-" & _
                  code & el(j).Num
        End If

        model = RemoveType(el(i).Model, code)
        maker = el(i).Maker


        ' =====================================
        ' ИЗМЕНЕНИЕ 5
        ' Модель сверху, изготовитель снизу
        ' =====================================

        If maker = "" Then
            Set names = SplitLines(model, 48)
        Else
            Set names = SplitLines(model, 32)
        End If

        Set makers = SplitLines(maker, 20)

        n = names.Count + makers.Count

        If n = 0 Then n = 1

        If n > 27 Then
            MsgBox "Слишком длинная запись: " & ref
            Exit Sub
        End If


        ' =====================================
        ' Групповой заголовок
        ' =====================================

        group = GroupName(code)

        If group <> oldGroup Then

            If r + n + 2 > maxRow Then
                NewPage wb, dst, page, r, maxRow
            End If

            ' Пустая строка перед заголовком
            r = r + 1

            dst.Cells(r, 2).Value = group
            dst.Range(dst.Cells(r, 2), _
                      dst.Cells(r, 3)).Merge

            ' Пустая строка после заголовка
            r = r + 2

            oldGroup = group

        ElseIf r + n - 1 > maxRow Then

            NewPage wb, dst, page, r, maxRow

        End If


        ' =====================================
        ' Запись элемента
        ' =====================================

        dst.Cells(r, 1).Value = ref

        ' Сначала модель в левой части

        For k = 1 To names.Count

            dst.Cells(r + k - 1, 2).Value = names(k)

        Next k

        ' Затем изготовитель/ТУ справа
        ' ИЗМЕНЕНИЕ 6

        For k = 1 To makers.Count

            dst.Cells(r + names.Count + k - 1, 3).Value = _
                makers(k)

        Next k

        ' Если фирмы нет - объединяем графы

        If maker = "" Then

            For k = 0 To n - 1

                dst.Range(dst.Cells(r + k, 2), _
                          dst.Cells(r + k, 3)).Merge

            Next k

        End If


        ' =====================================
        ' ИЗМЕНЕНИЕ 7
        ' Количество на последней строке
        ' =====================================

        dst.Cells(r + n - 1, 4).Value = j - i + 1

        r = r + n
        i = j + 1

    Loop

    MsgBox "Перечень готов. Страниц: " & page

End Sub


' =====================================
' ИЗМЕНЕНИЕ 8
' РАЗДЕЛЕНИЕ МОДЕЛИ И ИЗГОТОВИТЕЛЯ/ТУ
' =====================================

Sub SplitName(s As String, _
              ByRef model As String, _
              ByRef maker As String)

    Dim p As Long, word As String

    s = Trim(s)
    model = s
    maker = ""

    ' Поиск фирмы

    p = InStr(1, s, " ф. ", vbTextCompare)

    If p > 0 Then

        model = Trim(Left(s, p - 1))
        maker = Trim(Mid(s, p + 1))

    Else

        ' Поиск последнего слова с ТУ

        p = InStrRev(s, " ")

        If p > 0 Then

            word = Trim(Mid(s, p + 1))

            If Len(word) > 2 And _
               Right(UCase(word), 2) = "ТУ" And _
               InStr(word, ".") > 0 Then

                model = Trim(Left(s, p - 1))
                maker = word

            End If

        End If

    End If

End Sub


' =====================================
' УДАЛЕНИЕ ПОВТОРЯЮЩЕГОСЯ ТИПА
' =====================================

Function RemoveType(s As String, _
                    code As String) As String

    Dim prefix As String

    prefix = ""

    Select Case code

        Case "C"
            prefix = "Конденсатор"

        Case "R"
            prefix = "Резистор"

        Case "L"
            prefix = "Катушка индуктивности"

        Case "VD"
            prefix = "Диод"

        Case "VT"
            prefix = "Транзистор"

        Case "DA", "DD", "U"
            prefix = "Микросхема"

    End Select

    If prefix <> "" Then

        If StrComp(Left(s, Len(prefix) + 1), _
           prefix & " ", vbTextCompare) = 0 Then

            s = Mid(s, Len(prefix) + 2)

        End If

    End If

    RemoveType = s

End Function


' =====================================
' ОФОРМЛЕНИЕ СТРАНИЦЫ
' =====================================

Sub PreparePage(ws As Worksheet, page As Long)

    Dim lastRow As Long

    If page = 1 Then
        lastRow = 24
    Else
        lastRow = 31
    End If

    ws.Name = "ПЭ_" & page

    ws.Columns("A").ColumnWidth = 11
    ws.Columns("B").ColumnWidth = 40
    ws.Columns("C").ColumnWidth = 24
    ws.Columns("D").ColumnWidth = 5.5
    ws.Columns("E").ColumnWidth = 24

    With ws.Range("A1:E" & lastRow)

        .Font.Name = "Arial"
        .Font.Size = 12
        .RowHeight = 22.5
        .VerticalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous

    End With

    ws.Range("B1:C1").Merge

    ws.Range("A1").Value = "Поз. обозначение"
    ws.Range("B1").Value = "Наименование"
    ws.Range("D1").Value = "Кол."
    ws.Range("E1").Value = "Примечание"

    ws.Rows(1).RowHeight = 36

    ws.Range("A1:E1").HorizontalAlignment = xlCenter
    ws.Range("A1:E1").WrapText = True

    ws.Columns("A").HorizontalAlignment = xlCenter
    ws.Columns("D").HorizontalAlignment = xlCenter

    With ws.PageSetup

        .PaperSize = xlPaperA4
        .Orientation = xlPortrait
        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = 1
        .PrintArea = "A1:E" & lastRow

    End With

End Sub


' =====================================
' СОЗДАНИЕ СЛЕДУЮЩЕЙ СТРАНИЦЫ
' =====================================

Sub NewPage(wb As Workbook, _
            ByRef ws As Worksheet, _
            ByRef page As Long, _
            ByRef r As Long, _
            ByRef maxRow As Long)

    page = page + 1

    Set ws = wb.Worksheets.Add( _
        After:=wb.Worksheets(wb.Worksheets.Count))

    PreparePage ws, page

    r = 2
    maxRow = 31

End Sub


' =====================================
' БУКВЕННОЕ ПОЗИЦИОННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefCode(s As String) As String

    Dim i As Long

    s = Trim(UCase(s))

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then
            Exit For
        End If

    Next i

    RefCode = Left(s, i - 1)

End Function


' =====================================
' ЧИСЛОВОЕ ПОЗИЦИОННОЕ ОБОЗНАЧЕНИЕ
' =====================================

Function RefNum(s As String) As Long

    Dim i As Long

    For i = 1 To Len(s)

        If Mid(s, i, 1) Like "#" Then

            RefNum = Val(Mid(s, i))
            Exit Function

        End If

    Next i

End Function


' =====================================
' НАИМЕНОВАНИЯ ГРУПП
' =====================================

Function GroupName(code As String) As String

    Select Case code

        Case "C"
            GroupName = "КОНДЕНСАТОРЫ"

        Case "R"
            GroupName = "РЕЗИСТОРЫ"

        Case "L"
            GroupName = "КАТУШКИ ИНДУКТИВНОСТИ"

        Case "VD"
            GroupName = "ДИОДЫ"

        Case "VT"
            GroupName = "ТРАНЗИСТОРЫ"

        Case "DA", "DD", "U"
            GroupName = "МИКРОСХЕМЫ"

        Case "X", "XP", "XS"
            GroupName = "СОЕДИНИТЕЛИ"

        Case "ZQ"
            GroupName = "РЕЗОНАТОРЫ"

        Case Else
            GroupName = "ЭЛЕМЕНТЫ " & code

    End Select

End Function


' =====================================
' ПЕРЕНОС ТЕКСТА ПО СТРОКАМ
' =====================================

Function SplitLines(s As String, _
                    maxLen As Long) As Collection

    Dim result As New Collection
    Dim words As Variant, word As Variant
    Dim line As String, piece As String

    words = Split(Trim(s), " ")

    For Each word In words

        piece = CStr(word)

        Do While Len(piece) > maxLen

            If line <> "" Then
                result.Add line
                line = ""
            End If

            result.Add Left(piece, maxLen)
            piece = Mid(piece, maxLen + 1)

        Loop

        If Len(line) + Len(piece) + 1 > maxLen Then

            If line <> "" Then result.Add line
            line = ""

        End If

        If line <> "" Then line = line & " "
        line = line & piece

    Next word

    If line <> "" Then result.Add line

    Set SplitLines = result

End Function
